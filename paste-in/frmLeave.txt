Option Explicit

' ================================================================
' LEAVE (v22) - two modes
'  Going on leave: books the absence (shown as "Leave" everywhere outside
'   the Absences sheet), locks in the Home Unit and a leave group (LV-####).
'  Returning: picks an open leave, works out the top-up = the home unit's
'   average Total Hours gained between the first day off and the return
'   (grade 2 and below, not on leave, with Hours History), shows every
'   number in a confirm, then logs ONE Adjustment (Ref LV-####, Reason
'   "Leave return") and closes the absence. Void Entry LV-#### undoes it.
'  Every control is built in code (new form: Insert > UserForm, name it
'  frmLeave, paste this).
' ================================================================

Private WithEvents mOptGo As MSForms.OptionButton
Private WithEvents mOptBack As MSForms.OptionButton
Private WithEvents mEmp As MSForms.ComboBox
Private WithEvents mLeave As MSForms.ComboBox
Private WithEvents mCalcBtn As MSForms.CommandButton
Private WithEvents mSave As MSForms.CommandButton
Private WithEvents mClose As MSForms.CommandButton
Private WithEvents mOverride As MSForms.TextBox

Private mGoCtls As Collection
Private mBackCtls As Collection
Private mLeaveIds As Variant
Private mCalc As LeaveCalc
Private mCalcOk As Boolean
Private mStartId As String

Private Const LBL_L As Long = 14
Private Const BOX_L As Long = 128
Private Const ROW_H As Long = 26

Private Function C(ByVal nm As String) As Object
    Set C = Me.Controls("lv_" & nm)
End Function

Private Function AddCtl(ByVal progId As String, ByVal nm As String, ByVal L As Single, ByVal T As Single, _
                        ByVal W As Single, ByVal H As Single, Optional ByVal grp As String = "") As Object
    Dim o As Object
    Set o = FormAddCtl(Me, progId, "lv_" & nm, L, T, W, H)              ' v27.1: the shared builder (modUI); the groups stay here
    If grp = "G" Then mGoCtls.Add o
    If grp = "B" Then mBackCtls.Add o
    Set AddCtl = o
End Function

Private Function AddLabel(ByVal nm As String, ByVal cap As String, ByVal L As Single, ByVal T As Single, _
                          ByVal W As Single, Optional ByVal grp As String = "") As Object
    Dim o As Object
    Set o = AddCtl("Forms.Label.1", nm, L, T, W, 15, grp)
    o.Caption = cap
    Set AddLabel = o
End Function

Private Function RowT(ByVal i As Long) As Single
    RowT = 40 + (i - 1) * ROW_H
End Function

Private Sub FillCombo(ByVal cb As Object, ByVal items As Variant)
    FormFillCombo cb, items                                              ' v27.1: modUI
End Sub

' CalMenuLeaveReturn: open straight on one leave booking
Public Sub StartReturn(ByVal absId As String)
    mStartId = absId
    On Error Resume Next
    mOptBack.Value = True
    SetMode
    SelectLeave absId
    On Error GoTo 0
End Sub

Private Sub SetMode()
    Dim o As Object, going As Boolean: going = mOptGo.Value
    For Each o In mGoCtls
        o.Visible = going
    Next o
    For Each o In mBackCtls
        o.Visible = Not going
    Next o
    Me.Caption = IIf(going, "Leave - going", "Leave - returning")
    mSave.Caption = IIf(going, "SAVE LEAVE", "SAVE TOP-UP")
    mCalcOk = False
End Sub

Private Sub mOptGo_Click(): SetMode: End Sub
Private Sub mOptBack_Click(): SetMode: End Sub

Private Sub LoadLeaves()
    Dim d As Variant, i As Long
    mLeave.Clear
    d = OpenLeaves(mLeaveIds)
    If IsArray(d) Then
        For i = LBound(d) To UBound(d)
            mLeave.AddItem d(i)
        Next i
    End If
    If mStartId <> "" Then SelectLeave mStartId
End Sub

Private Sub SelectLeave(ByVal absId As String)
    If Not IsArray(mLeaveIds) Then Exit Sub
    Dim i As Long
    For i = LBound(mLeaveIds) To UBound(mLeaveIds)
        If StrComp(CStr(mLeaveIds(i)), absId, vbTextCompare) = 0 Then mLeave.ListIndex = i: Exit Sub
    Next i
End Sub

Private Sub mEmp_Change()
    If mEmp.ListIndex < 0 Then C("hEmp").Caption = "": Exit Sub
    Dim hu As String: hu = HomeUnitOf(mEmp.Text)
    C("hEmp").Caption = IIf(hu = "", "NO HOME UNIT on the roster - set it with Employee > Edit first, the return top-up needs it", "home unit: " & hu)
    C("hEmp").ForeColor = IIf(hu = "", RGB(190, 30, 30), RGB(120, 120, 120))
End Sub

Private Sub mLeave_Change()
    mCalcOk = False
    C("result").Caption = "Enter the return date, then click Work out the top-up."
End Sub

' ---- the calculation ----
Private Sub mCalcBtn_Click()
    mCalcOk = False
    If mLeave.ListIndex < 0 Or Not IsArray(mLeaveIds) Then MsgBox "Pick who's back.", vbExclamation, "Call Out Roster": Exit Sub
    Dim t As String: t = Trim$(C("txtReturn").Text)
    Dim dRet As Date
    If Not ParseTypedDate(t, dRet) Then MsgBox "Enter the return date (first day back, e.g. Oct 12).", vbExclamation, "Call Out Roster": Exit Sub
    If dRet > Date Then                            ' v25 (D-C7): the top-up measures a gap that has happened
        MsgBox "The return date is after today. Log the return on or after the first day back.", vbExclamation, "Call Out Roster"
        Exit Sub
    End If
    Me.MousePointer = fmMousePointerHourGlass
    Dim ok As Boolean
    ok = ComputeLeaveReturn(CStr(mLeaveIds(mLeave.ListIndex)), dRet, mCalc)
    Me.MousePointer = fmMousePointerDefault
    If Not ok Then
        C("result").Caption = "Can't work it out: " & mCalc.Warnings
        C("result").ForeColor = RGB(190, 30, 30)
        Exit Sub
    End If
    C("result").ForeColor = RGB(30, 56, 100)
    C("result").Caption = mCalc.Summary & vbLf & "Peers: " & mCalc.PeerNames & _
                          IIf(mCalc.Warnings <> "", vbLf & "Check: " & mCalc.Warnings, "")
    mOverride.Text = ""
    C("txtWhy").Text = ""
    mCalcOk = True
End Sub

' ---- save ----
Private Sub mSave_Click()
    If mOptGo.Value Then SaveGoing Else SaveReturning
End Sub

Private Sub SaveGoing()
    Dim problems As String, d1 As Date, d2 As Variant
    If mEmp.ListIndex < 0 Then problems = problems & "- Employee" & vbNewLine
    If C("cboType").ListIndex < 0 Then problems = problems & "- Type" & vbNewLine
    Dim dExp As Date                               ' v25: every date through the one parser
    If Not ParseTypedDate(C("txtFirst").Text, d1) Then problems = problems & "- First day off" & vbNewLine
    If Trim$(C("txtExp").Text) <> "" And Not ParseTypedDate(C("txtExp").Text, dExp) Then problems = problems & "- Expected return (a date, or blank)" & vbNewLine
    If problems <> "" Then MsgBox "Please fill in:" & vbNewLine & vbNewLine & problems, vbExclamation, "Leave": Exit Sub
    If Trim$(C("txtExp").Text) <> "" Then
        d2 = dExp
        If d2 <= d1 Then MsgBox "The expected return must be after the first day off.", vbExclamation, "Call Out Roster": Exit Sub
    Else
        d2 = Empty
    End If
    If HomeUnitOf(mEmp.Text) = "" Then
        If MsgBox(mEmp.Text & " has no Home Unit on the roster. The return top-up needs it. Save the leave anyway " & _
                  "(set it with Employee > Edit before they return)?", vbYesNo + vbExclamation, "Leave") = vbNo Then Exit Sub
    End If
    Dim hu As String, id As String
    id = SaveLeaveGoing(mEmp.Text, C("cboType").Text, d1, d2, Trim$(C("txtNote").Text), hu)
    If id = "" Then Exit Sub
    RefreshCalendar
    C("status").Caption = "Saved: " & mEmp.Text & " on leave from " & Format$(d1, "mmm d") & _
                          IIf(IsDate(d2), ", back " & Format$(d2, "mmm d"), ", open-ended") & ". Home unit " & IIf(hu = "", "(none)", hu) & ".  (" & id & ")"
    LoadLeaves
    ' v27 (item 7): the receipt card
    ShowReceiptNow "Leave " & id & " saved", _
                   "  " & id & "  Leave going  " & mEmp.Text & "  from " & Format$(d1, "ddd mmm d") & IIf(IsDate(d2), " to " & Format$(CDate(d2), "ddd mmm d"), " (open-ended)") & "  (an Absences row; no OT Log row yet)" & vbNewLine, _
                   "  Nobody's Total Hours changed. The return top-up is worked out when they come back (Leave: returning), from home unit " & IIf(hu = "", "(none - set it with Employee > Edit)", hu) & "." & vbNewLine, _
                   "  Calendar: Leave on every scheduled shift; the Absences sheet (admin); Needs Attention lists the open leave." & vbNewLine & _
                   "  The members' file shows Leave after the next update (never the type)." & vbNewLine, _
                   "  Book Time Off > Void this booking (admin) voids it." & vbNewLine
End Sub

Private Sub SaveReturning()
    If Not mCalcOk Then MsgBox "Click Work out the top-up first.", vbExclamation, "Call Out Roster": Exit Sub
    Dim hrs As Double: hrs = mCalc.Average
    Dim why As String: why = Trim$(C("txtWhy").Text)
    If Trim$(mOverride.Text) <> "" Then
        If Not IsNumeric(Trim$(mOverride.Text)) Then MsgBox "The override must be a number of hours.", vbExclamation, "Call Out Roster": Exit Sub
        hrs = CDbl(Trim$(mOverride.Text))
        If why = "" Then MsgBox "An override needs a reason (it goes in the OT Log).", vbExclamation, "Call Out Roster": Exit Sub
    ElseIf mCalc.NPeers = 0 Then
        MsgBox "No peers were found, so the average is 0. Enter an override with a reason, or fix the Home Unit / Hours History first.", _
               vbExclamation, "Leave"
        Exit Sub
    End If
    Dim msg As String
    msg = mCalc.Emp & " returns " & Format$(mCalc.ReturnDate, "ddd mmm d, yyyy") & vbNewLine & vbNewLine & mCalc.Summary & vbNewLine
    If hrs <> mCalc.Average Then msg = msg & vbNewLine & "OVERRIDE: add " & Format$(hrs, "0.0") & " h instead (" & why & ")" & vbNewLine
    msg = msg & vbNewLine & "This logs one Adjustment (Ref LV-####, Reason 'Leave return'" & IIf(hrs = 0, ", 0 h: the return goes on record", "") & _
          ") and closes the leave. Void Entry undoes it." & vbNewLine & vbNewLine & "Save?"
    If MsgBox(msg, vbOKCancel + vbQuestion, "Confirm leave return") <> vbOK Then Exit Sub
    Dim before As Object: Set before = ReceiptBefore()        ' v27 (item 7)
    Dim id As String: id = SaveLeaveReturn(mCalc, hrs, why)
    If id = "" Then Exit Sub
    RefreshCalendar
    C("status").Caption = "Saved: " & Format$(hrs, "0.0") & " h added to " & mCalc.Emp & "; leave closed.  (" & id & ")"
    mCalcOk = False
    LoadLeaves
    ShowReceiptNow "Leave return " & id & " saved", _
                   "  " & id & "  Adjustment (Leave return)  " & mCalc.Emp & "  " & Format$(hrs, "0.0") & " h" & IIf(why <> "", "  (" & why & ")", "") & vbNewLine & _
                   "  " & Replace(mCalc.Summary, vbNewLine, vbNewLine & "  ") & vbNewLine, ReceiptChanges(before), _
                   "  OT Log: the Adjustment row " & id & " with the calculation in its Note. Roster: " & mCalc.Emp & "'s Total Hours. Calendar: the leave is closed." & vbNewLine & _
                   "  The members' file shows it after the next update." & vbNewLine, WrongText()
End Sub

Private Sub mClose_Click()
    Unload Me
End Sub
