Private Function AddCtl(ByVal progId As String, ByVal nm As String, ByVal L As Single, ByVal T As Single, _
                        ByVal W As Single, ByVal H As Single, Optional ByVal grp As String = "") As Object

Private Function AddCombo(ByVal nm As String, ByVal L As Single, ByVal T As Single, ByVal W As Single, _

Private Function RowT(ByVal i As Long) As Single
    RowT = 40 + (i - 1) * ROW_H
End Function

Private Sub FillCombo(ByVal cb As Object, ByVal items As Variant)
    Dim i As Long
    cb.Clear
    If Not IsArray(items) Then Exit Sub
    For i = LBound(items) To UBound(items)
        cb.AddItem CStr(items(i))
    Next i
End Sub

Private Function ModeKey() As String
    If mOptMove.Value Then
        ModeKey = "M"
    ElseIf mOptRename.Value Then
        ModeKey = "R"
    ElseIf mOptRemove.Value Then
        ModeKey = "X"
    ElseIf mOptBack.Value Then
        ModeKey = "B"
    ElseIf mOptEdit.Value Then
        ModeKey = "E"
    Else
        ModeKey = "A"
    End If
End Function

' v26.5: a double-click on a roster row (modPeople.OpenEmployeeEdit) opens straight on Edit for one person
Public Sub StartEdit(ByVal emp As String)
    On Error Resume Next
    mOptEdit.Value = True
    SetMode
    SelectIn mEditEmp, emp                       ' its Change event fills the boxes
    On Error GoTo 0
End Sub

' Edit: the boxes follow the person picked
Private Sub mEditEmp_Change()
    FillEdit
End Sub

Private Sub FillEdit()
    Dim u As Long
    mEdHome = "": mEdCrew = "": mEdPhone = "": mEdEmail = "": mEdSen = ""
    ReDim mEdCodes(0 To mEdN - 1)
    C("eHEmp").Caption = ""
    If mEditEmp.ListIndex < 0 Then
        C("eHome").ListIndex = -1: C("eCrew").ListIndex = -1
        C("ePhone").Text = "": C("eEmail").Text = "": C("eSen").Text = ""
        For u = 0 To mEdN - 1
            C("eC" & u).Text = ""
        Next u
        Exit Sub
    End If
    Dim codes() As String
    If Not PersonDetails(mEditEmp.Text, mEdHome, mEdCrew, mEdPhone, mEdEmail, mEdSen, codes) Then Exit Sub
    C("eHEmp").Caption = PersonSummary(mEditEmp.Text)
    C("eHome").ListIndex = -1: C("eCrew").ListIndex = -1    ' a blank or unknown value must not keep the previous person's
    SelectIn C("eHome"), mEdHome
    SelectIn C("eCrew"), mEdCrew
    C("ePhone").Text = mEdPhone
    C("eEmail").Text = mEdEmail
    C("eSen").Text = mEdSen
    C("eFrom").Text = Format$(Date, "mmm d, yyyy")
    For u = 0 To mEdN - 1
        mEdCodes(u) = codes(u)
        C("eC" & u).Text = codes(u)
    Next u
End Sub

Private Sub SelectIn(ByVal cb As Object, ByVal txt As String)
    Dim i As Long
    For i = 0 To cb.ListCount - 1
        If StrComp(CStr(cb.List(i)), txt, vbTextCompare) = 0 Then cb.ListIndex = i: Exit Sub
    Next i
End Sub

Private Sub SetMode()
    Dim key As String: key = ModeKey()
    Dim k As Variant, o As Object, bag As Collection
    For Each k In mGroups.Keys                  ' hide every group, then show this mode's
        Set bag = mGroups(k)
        For Each o In bag
            o.Visible = False
        Next o
    Next k
    Set bag = mGroups(key)
    For Each o In bag
        o.Visible = True
    Next o
    Select Case key
        Case "A": Me.Caption = "Employee - add": mSave.Caption = "ADD EMPLOYEE"
        Case "M": Me.Caption = "Employee - move to unit": mSave.Caption = "SAVE MOVE"
        Case "R": Me.Caption = "Employee - rename": mSave.Caption = "RENAME"
        Case "X": Me.Caption = "Employee - remove": mSave.Caption = "REMOVE FROM ROSTER"
        Case "B": Me.Caption = "Employee - bring back": mSave.Caption = "BRING BACK"
        Case "E": Me.Caption = "Employee - edit details": mSave.Caption = "SAVE CHANGES"
    End Select
    C("result").Caption = IIf(key = "A" Or key = "M", "Fill in the boxes, then click Work out the hours. Nothing is saved until you click " & mSave.Caption & ".", "")
    C("result").ForeColor = RGB(30, 56, 100)
End Sub

' Move: show what the person holds now and list their codes (Home Unit's code ticked)
Private Sub mMoveEmp_Change()
    Dim lb As Object: Set lb = C("mClear")
    lb.Clear
    mMoveN = 0
    If mMoveEmp.ListIndex < 0 Then C("mHEmp").Caption = "": Exit Sub
    C("mHEmp").Caption = PersonSummary(mMoveEmp.Text)
    Dim codes() As String, i As Long, hu As String
    mMoveN = PersonCodes(mMoveEmp.Text, mMoveUnits, codes)
    hu = HomeUnitOf(mMoveEmp.Text)
    For i = 1 To mMoveN
        lb.AddItem mMoveUnits(i) & "   (" & codes(i) & ")"
        If StrComp(mMoveUnits(i), hu, vbTextCompare) = 0 Then lb.Selected(i - 1) = True
    Next i
End Sub

Private Function ClearList() As String          ' "Unit A|Unit B" from the ticked codes
    Dim lb As Object: Set lb = C("mClear")
    Dim i As Long, s As String
    For i = 1 To mMoveN
        If lb.Selected(i - 1) Then s = s & IIf(s = "", "", "|") & mMoveUnits(i)
    Next i
    ClearList = s
End Function

Private Function PickedCrew() As String         ' "" = no change

End Function


' ================================================================
' THE CALCULATION (shown before anything is saved)
' ================================================================
Private Function CalcAdd(ByRef pc As PeopleCalc, ByRef startDate As Date) As Boolean
    If C("aUnit").ListIndex < 0 Then MsgBox "Pick the home unit.", vbExclamation, "Add employee": Exit Function
    If Not ParseTypedDate(C("aStart").Text, startDate) Then
        MsgBox "Enter the start date, e.g. " & Format$(Date, "mmm d, yyyy") & ".", vbExclamation, "Add employee"
        Exit Function
    End If
    Me.MousePointer = fmMousePointerHourGlass
    CalcAdd = ComputeNewHire(C("aUnit").Text, startDate, Trim$(C("aName").Text), pc)
    Me.MousePointer = fmMousePointerDefault
    ShowCalc CalcAdd, pc
End Function

Private Function CalcMove(ByRef pc As PeopleCalc) As Boolean
    Dim eff As Date
    If mMoveEmp.ListIndex < 0 Then MsgBox "Pick the person.", vbExclamation, "Move to unit": Exit Function
    If C("mUnit").ListIndex < 0 Then MsgBox "Pick the new unit.", vbExclamation, "Move to unit": Exit Function
    If Not ParseTypedDate(C("mDate").Text, eff) Then
        MsgBox "Enter the effective date, e.g. " & Format$(Date, "mmm d, yyyy") & ".", vbExclamation, "Move to unit"
        Exit Function
    End If
    Dim had As String, hadOn As Date, hadUnit As String
    had = PlannedFor(mMoveEmp.Text, hadOn, hadUnit)          ' v24.3: one move at a time per person
    If had <> "" Then
        MsgBox mMoveEmp.Text & " already has a planned move: " & had & ", to " & hadUnit & " on " & Format$(hadOn, "mmm d, yyyy") & "." & vbNewLine & vbNewLine & _
               "To change it, cancel it first (Needs Attention > Planned unit moves, double-click the line), then enter it again.", _
               vbExclamation, "Move to unit"
        Exit Function
    End If
    Me.MousePointer = fmMousePointerHourGlass
    CalcMove = ComputeUnitMove(mMoveEmp.Text, C("mUnit").Text, eff, pc, True)
    Me.MousePointer = fmMousePointerDefault
    ShowCalc CalcMove, pc
End Function

Private Sub ShowCalc(ByVal ok As Boolean, ByRef pc As PeopleCalc)
    If Not ok Then
        C("result").Caption = "Can't work it out: " & pc.Warnings
        C("result").ForeColor = RGB(190, 30, 30)
        Exit Sub
    End If
    C("result").ForeColor = RGB(30, 56, 100)
    C("result").Caption = pc.Summary & IIf(pc.PeerNames <> "", vbLf & "Peers: " & pc.PeerNames, "") & _
                          IIf(pc.Warnings <> "", vbLf & "Check: " & pc.Warnings, "")
End Sub

Private Sub mCalcBtn_Click()
    Dim pc As PeopleCalc, d As Date
    If ModeKey() = "A" Then
        CalcAdd pc, d
    ElseIf ModeKey() = "M" Then
        CalcMove pc
    End If
End Sub

' v26.5: Edit - what differs from the boxes' starting values is listed, confirmed, then saved by
' modPeople.SaveEmployeeEdit (events off, the Grade Log, the crew history, the totals check)
Private Sub SaveEdit()
    If mEditEmp.ListIndex < 0 Then MsgBox "Pick the person.", vbExclamation, "Edit employee": Exit Sub
    Dim emp As String: emp = mEditEmp.Text
    Dim homeUnit As String, crewTxt As String, crewFrom As Date, sen As Variant, senDate As Date, u As Long
    Dim codes() As String: ReDim codes(0 To mEdN - 1)
    Dim what As String
    If C("eHome").ListIndex >= 0 Then homeUnit = C("eHome").Text
    If homeUnit <> "" And StrComp(homeUnit, mEdHome, vbTextCompare) <> 0 Then what = what & "- home unit: " & mEdHome & " -> " & homeUnit & vbNewLine
    If C("eCrew").ListIndex >= 0 Then crewTxt = C("eCrew").Text
    If crewTxt <> "" And StrComp(NormCrew(crewTxt), NormCrew(mEdCrew), vbTextCompare) <> 0 Then
        If Not ParseTypedDate(C("eFrom").Text, crewFrom) Then
            MsgBox "Enter the date the crew change starts, e.g. " & Format$(Date, "mmm d, yyyy") & ".", vbExclamation, "Edit employee"
            Exit Sub
        End If
        what = what & "- crew: " & mEdCrew & " -> " & crewTxt & " from " & Format$(crewFrom, "ddd mmm d, yyyy") & vbNewLine
    End If
    If Trim$(C("ePhone").Text) <> mEdPhone Then what = what & "- phone: " & IIf(mEdPhone = "", "(none)", mEdPhone) & " -> " & IIf(Trim$(C("ePhone").Text) = "", "(none)", Trim$(C("ePhone").Text)) & vbNewLine
    If Trim$(C("eEmail").Text) <> mEdEmail Then what = what & "- email: " & IIf(mEdEmail = "", "(none)", mEdEmail) & " -> " & IIf(Trim$(C("eEmail").Text) = "", "(none)", Trim$(C("eEmail").Text)) & vbNewLine
    sen = ""
    If Trim$(C("eSen").Text) <> "" And Trim$(C("eSen").Text) <> mEdSen Then     ' unchanged text (even a text value the roster holds) is left alone
        If Not ParseFullDate(C("eSen").Text, senDate) Then
            MsgBox "The seniority date needs the year, e.g. Mar 3, 2011.", vbExclamation, "Edit employee"
            Exit Sub
        End If
        sen = senDate
        If Format$(senDate, "mmm d, yyyy") <> mEdSen Then what = what & "- seniority date: " & IIf(mEdSen = "", "(none)", mEdSen) & " -> " & Format$(senDate, "mmm d, yyyy") & vbNewLine
    End If
    For u = 0 To mEdN - 1
        codes(u) = Trim$(C("eC" & u).Text)
        If StrComp(codes(u), mEdCodes(u), vbBinaryCompare) <> 0 Then
            what = what & "- " & DeptOfCol(UnitColFirst() + u) & " code: " & IIf(mEdCodes(u) = "", "(none)", mEdCodes(u)) & " -> " & IIf(codes(u) = "", "(cleared)", codes(u)) & vbNewLine
        End If
    Next u
    If what = "" Then MsgBox "Nothing has changed for " & emp & ".", vbInformation, "Edit employee": Exit Sub
    If MsgBox("Save these changes for " & emp & "?" & vbNewLine & vbNewLine & what & vbNewLine & _
              "Code changes go in the Grade Log; a crew change goes in the crew history from its date. Nobody's hours change.", _
              vbOKCancel + vbQuestion, "Confirm changes") <> vbOK Then Exit Sub
    Dim report As String
    Me.MousePointer = fmMousePointerHourGlass
    Dim ok As Boolean: ok = SaveEmployeeEdit(emp, homeUnit, crewTxt, crewFrom, C("ePhone").Text, C("eEmail").Text, sen, codes, report)
    Me.MousePointer = fmMousePointerDefault
    If Not ok Then Exit Sub
    FillEdit                                     ' the boxes show what is saved now
    Finished report, "Edit employee"
End Sub

Private Sub Finished(ByVal report As String, ByVal title As String)
    C("status").Caption = Split(report, vbNewLine)(0)
    ' the lists follow the roster
    Dim keepMove As String: keepMove = mMoveEmp.Text
    Dim keepEdit As String: keepEdit = mEditEmp.Text
    FillCombo mMoveEmp, RosterNames()
    FillCombo mEditEmp, RosterNames()
    If keepEdit <> "" Then SelectIn mEditEmp, keepEdit
    FillCombo C("rEmp"), RosterNames()
    FillCombo C("xEmp"), RosterNames()
    FillCombo C("bEmp"), FormerNames()
    C("mClear").Clear: mMoveN = 0: C("mHEmp").Caption = ""
    C("rNew").Text = ""
    If keepMove <> "" Then SelectIn mMoveEmp, keepMove
    MsgBox report, vbInformation, title
End Sub

Private Sub SaveAdd()
    Dim problems As String, nm As String, sen As Variant, senDate As Date
    nm = Trim$(C("aName").Text)
    If nm = "" Then problems = problems & "- Name" & vbNewLine
    If C("aCrew").ListIndex < 0 Then problems = problems & "- Crew" & vbNewLine
    If C("aUnit").ListIndex < 0 Then problems = problems & "- Home unit" & vbNewLine
    If Trim$(C("aCode").Text) = "" Then problems = problems & "- Grade code" & vbNewLine
    If Trim$(C("aSen").Text) = "" Then
        problems = problems & "- Seniority date" & vbNewLine
    ElseIf Not ParseFullDate(C("aSen").Text, senDate) Then
        problems = problems & "- Seniority date (with the year, e.g. Mar 3, 2011)" & vbNewLine
    End If
    If problems <> "" Then MsgBox "Please fill in:" & vbNewLine & vbNewLine & problems, vbExclamation, "Add employee": Exit Sub
    sen = senDate

    Dim pc As PeopleCalc, startDate As Date
    If Not CalcAdd(pc, startDate) Then Exit Sub
    Dim hrs As Double: hrs = pc.AddHours
    Dim why As String: why = Trim$(C("aWhy").Text)
    If Trim$(C("aOver").Text) <> "" Then
        If Not IsNumeric(Trim$(C("aOver").Text)) Then MsgBox "The override must be a number of hours.", vbExclamation, "Add employee": Exit Sub
        hrs = CDbl(Trim$(C("aOver").Text))
        If hrs < 0 Then MsgBox "The starting hours can't be negative.", vbExclamation, "Add employee": Exit Sub
        If why = "" Then MsgBox "An override needs a reason (it goes in the OT Log).", vbExclamation, "Add employee": Exit Sub
    End If

    Dim msg As String
    msg = "Add " & nm & "?" & vbNewLine & vbNewLine & _
          "Crew " & C("aCrew").Text & "   |   " & C("aUnit").Text & " " & Trim$(C("aCode").Text) & vbNewLine & _
          "Seniority " & Format$(senDate, "mmm d, yyyy") & "   |   starts " & Format$(startDate, "ddd mmm d, yyyy") & vbNewLine & vbNewLine & _
          pc.Summary & vbNewLine
    If pc.Warnings <> "" Then msg = msg & "Check: " & pc.Warnings & vbNewLine
    If hrs <> pc.AddHours Then msg = msg & vbNewLine & "OVERRIDE: start at " & Format$(hrs, "0.0") & " h instead (" & why & ")" & vbNewLine
    msg = msg & vbNewLine & "This adds the roster row and logs " & _
          IIf(hrs = 0, "no hours.", "one Adjustment (Ref NH-####, Reason 'New hire'). Void Entry undoes the hours.")
    If MsgBox(msg, vbOKCancel + vbQuestion, "Confirm new employee") <> vbOK Then Exit Sub

    Dim report As String
    If Not SaveNewHire(nm, C("aPhone").Text, C("aEmail").Text, C("aCrew").Text, sen, C("aUnit").Text, _
                       C("aCode").Text, startDate, pc, hrs, why, report) Then Exit Sub
    C("aName").Text = "": C("aPhone").Text = "": C("aEmail").Text = "": C("aSen").Text = ""
    C("aOver").Text = "": C("aWhy").Text = ""
    C("result").Caption = ""
    Finished report, "Add employee"
End Sub

Private Sub SaveMove()
    Dim pc As PeopleCalc
    If Trim$(C("mCode").Text) = "" Then MsgBox "Enter the grade code in the new unit (e.g. D2).", vbExclamation, "Move to unit": Exit Sub
    If Not CalcMove(pc) Then Exit Sub
    Dim clearTxt As String: clearTxt = ClearList()
    Dim crewTxt As String: crewTxt = PickedCrew()
    Dim msg As String
    If pc.AsOf > Date Then                      ' v24.3: a date after today plans the move
        SavePlanned pc, clearTxt, crewTxt
        Exit Sub
    End If
    msg = pc.Summary & vbNewLine & vbNewLine & _
          "New code: " & Trim$(C("mCode").Text) & " in " & pc.Unit & " (Home Unit becomes " & pc.Unit & ")" & vbNewLine & _
          "Codes cleared: " & IIf(clearTxt = "", "none", Replace(clearTxt, "|", ", ")) & vbNewLine & _
          "Crew: " & IIf(crewTxt = "", "no change", "changes to " & crewTxt & " from " & Format$(pc.AsOf, "mmm d, yyyy")) & vbNewLine
    If pc.Warnings <> "" Then msg = msg & "Check: " & pc.Warnings & vbNewLine
    Dim posts As String
    If clearTxt <> "" Then posts = ExemptFromOpenPostings(pc.Emp, pc.AsOf, "|" & clearTxt & "|", "", True)
    If posts <> "" Then msg = msg & "Open posting(s) in the old unit where they will not be charged: " & posts & vbNewLine
    If clearTxt = "" Then msg = msg & vbNewLine & "NOTE: no code is being cleared. If this is only an extra sign-off, Cancel and add the code with Edit instead." & vbNewLine
    msg = msg & vbNewLine & IIf(pc.AddHours > 0, "This logs one Adjustment (Ref UM-####, Reason 'Unit move'). Void Entry undoes the hours.", "No hours are logged.") & _
          vbNewLine & "The code changes go in the Grade Log." & vbNewLine & vbNewLine & "Save?"
    If MsgBox(msg, vbOKCancel + vbQuestion, "Confirm unit move") <> vbOK Then Exit Sub

    Dim report As String
    If Not SaveUnitMove(pc.Emp, pc.Unit, C("mCode").Text, clearTxt, crewTxt, pc, report) Then Exit Sub
    C("mCode").Text = "": C("mCrew").ListIndex = 0
    C("result").Caption = ""
    Finished report, "Move to unit"
End Sub

' v24.3: the move is dated after today. Nothing on the roster changes now;
' SavePlannedMove writes one row and the move is applied on the day.
Private Sub SavePlanned(ByRef pc As PeopleCalc, ByVal clearTxt As String, ByVal crewTxt As String)
    Dim msg As String, posts As String, report As String
    Dim code As String: code = Trim$(C("mCode").Text)
    msg = "PLANNED MOVE - nothing changes on the roster today." & vbNewLine & vbNewLine & _
          pc.Emp & " moves to " & pc.Unit & " (" & code & ") on " & Format$(pc.AsOf, "ddd mmm d, yyyy") & "." & vbNewLine & _
          "Codes cleared on that day: " & IIf(clearTxt = "", "none", Replace(clearTxt, "|", ", ")) & vbNewLine & _
          "Crew: " & IIf(crewTxt = "", "no change", "changes to " & crewTxt & " from that date (column E shows it from today; lists use the old crew until then)") & vbNewLine & vbNewLine
    msg = msg & "For shifts on or after that date they are on " & pc.Unit & "'s call lists" & _
          IIf(clearTxt = "", "", " and off " & Replace(clearTxt, "|", ", ") & "'s") & ". Until the move is applied they are listed in " & _
          pc.Unit & " at the hours they would have after the top-up." & vbNewLine & vbNewLine
    msg = msg & pc.Summary & vbNewLine
    If pc.Warnings <> "" Then msg = msg & "Check: " & pc.Warnings & vbNewLine
    If clearTxt <> "" Then posts = ExemptFromOpenPostings(pc.Emp, pc.AsOf, "|" & clearTxt & "|", "", True)
    If posts <> "" Then msg = msg & "Open posting(s) in the old unit where they will not be charged: " & posts & vbNewLine
    If clearTxt = "" Then msg = msg & vbNewLine & "NOTE: no code is being cleared, so they stay on their present lists as well." & vbNewLine
    msg = msg & vbNewLine & "The move is applied the first time an admin opens the file on or after that date." & vbNewLine & _
          "To change or cancel it: Needs Attention > Planned unit moves." & vbNewLine & vbNewLine & _
          "A backup copy is saved first. Save the planned move?"
    If MsgBox(msg, vbOKCancel + vbQuestion, "Confirm planned move") <> vbOK Then Exit Sub

    If Not SavePlannedMove(pc.Emp, pc.Unit, code, clearTxt, crewTxt, pc.AsOf, report) Then Exit Sub
    C("mCode").Text = "": C("mCrew").ListIndex = 0
    C("result").Caption = ""
    Finished report, "Move to unit"
End Sub
