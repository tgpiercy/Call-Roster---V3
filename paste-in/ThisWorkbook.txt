Option Explicit
' @version v29.0.3 (Oct 28)

Private mRestoreAdmin As Boolean
Private mWasActive As Object
Private mQuietSeen As String           ' v24.2: background events already logged once this session
Private mRestoreDesk As Boolean        ' v25.3: the Desk view was on before the save's view switch

' Opening: protect every sheet (macros can still edit), then show the
' admin view only to people on ADMIN_USERS. No password prompt.
Private Sub Workbook_Open()
    Dim k As Long
    Application.ScreenUpdating = False
    For k = 1 To 14                    ' v24.2: a step that fails is written to the Error Log; the rest still run
        OpenStep k
    Next k
    On Error Resume Next
    Application.ScreenUpdating = True
    ThisWorkbook.Saved = True
    RecordDiskStamp                    ' v26.5.5 (A2): the file's on-disk time; a save finds it changed = someone else saved meanwhile
    OpenChecks                         ' v20: read-only notice (who has it), AutoSave off on SharePoint
    SetShortcuts True                  ' v25.5: Ctrl+Shift+S / C / K / D while this workbook is active (modUI)
    FirstOpenOnOpen                    ' v27 (item 20): a login's very FIRST open - the three things that matter (modSystem); before the what's-new box, which it reads to tell an old login from a new one
    WhatsNewOnOpen                     ' v25.5: the what's-new box once per login per release (modSystem) - last, on a painted screen (review)
    ThisWorkbook.Saved = True          ' the hidden name it may write goes with the next real save
    On Error GoTo 0
End Sub

Private Sub OpenStep(ByVal k As Long)
    On Error GoTo Oops
    Select Case k
        Case 1: BackupOnOpen           ' v21: first admin open of the day saves a dated copy (never blocks)
        Case 2: ProtectAll
        Case 3: ApplyView IsAdmin(), IsDeskUser()      ' v25.3: a Desk user gets the Desk view (modDesk)
        Case 4: EnsureDataThrough: EnsureRefusedThrough   ' v21: the provisional-OT formulas need the data-through name; v26.5: Refused counts the periods up to the current one
        Case 5: GradeLogOnOpen         ' v21: grade-change snapshot (and anything changed with macros off)
        Case 6: PeopleOnOpen           ' v24: crew history (new names, a crew changed with macros off)
        Case 7: HolidaysOnOpen         ' v20: this year's and next year's holidays listed (admins)
        Case 8: CalOnOpen              ' calendar: this month, fresh needs-attention box
        Case 9: IntegrityOnOpen        ' v23: quiet integrity check for admins (result on the status line)
        Case 10: RequestsOnOpen           ' v28 (item 1): an admin's open folds the members' request files into the Requests table (modRequests) - BEFORE the Desk draws its tile
        Case 11: DeskOpenLanding          ' v25.3: a Desk user lands on the Desk, everyone else on the roster (modDesk)
        Case 12: PracticeBannerOnOpen     ' v26.5.5: a practice copy carries a red banner on every sheet and in its title (modSystem)
        Case 13: ExportTablesOnOpen       ' v27 (item 25): the day's CSV export of every table beside the backups (modEvidence)
        Case 14: OpensOnOpen              ' v27 (item 32): one row on the admin-only Opens sheet (modSystem)
    End Select
    Exit Sub
Oops:
    LogError "Workbook_Open step " & k, Err.Number, Err.Description
End Sub

' v24.2: an error in a background event is logged once per session, with no message
Private Sub Quietly(ByVal what As String)
    Dim en As Long: en = Err.Number
    Dim em As String: em = Err.Description
    On Error Resume Next                            ' never raises
    If InStr(mQuietSeen, "|" & what & "|") > 0 Then Exit Sub
    mQuietSeen = mQuietSeen & "|" & what & "|"
    LogError what, en, em
End Sub

' SystemCheck reads this to confirm ThisWorkbook was re-pasted
Public Function CodeVersion() As String
    CodeVersion = "ThisWorkbook v29.0.3 (Oct 28)"
End Function

' Every save (manual, after a form submit, or on close) stores the
' RESTRICTED view - phones hidden, admin sheets hidden, all protected -
' then puts the admin view back. Blocks saving after a failed migration.
Private Sub Workbook_BeforeSave(ByVal SaveAsUI As Boolean, Cancel As Boolean)
    If NoSave() Then
        Cancel = True
        MsgBox "Saving is blocked: a check failed earlier in this session, or the file changed on disk while you had it open." & vbNewLine & _
               "Close this workbook WITHOUT saving and reopen it (after a failed conversion: restore from the backup).", vbCritical, "Not saved"
        Exit Sub
    End If
    Dim savedAt As Date                            ' v26.5.5 (A2): someone else saved this file since it was opened here
    If StaleOnDisk(savedAt) Then
        Cancel = True
        MarkNoSave
        MsgBox "Someone else saved this file at " & Format$(savedAt, "ddd mmm d h:mm AM/PM") & " while you had it open." & vbNewLine & vbNewLine & _
               "Close WITHOUT saving, reopen, and log your entry again. Nothing you did here was written to the file.", _
               vbCritical, "Not saved - the file changed"
        Exit Sub
    End If
    On Error Resume Next
    HideCalloutPhonesForSave                       ' v27 (item 16): a supervisor's printed roster loses its phone column before the file is saved (review L8: before the backup copy too)
    mRestoreAdmin = gAdminUnlocked
    mRestoreDesk = gDeskMode                       ' v25.3
    If mRestoreAdmin Then
        Set mWasActive = ActiveSheet
        Application.ScreenUpdating = False          ' restored by Workbook_AfterSave (the admin view comes back there)
        ApplyView False
    End If
    ' v29.0.3 (review HIGH 3.1): the hourly backup is taken AFTER the view switch, so it is saved in the RESTRICTED view
    ' like the daily backup and every copy - before, an admin's save wrote a backup with phones, emails, Payroll #,
    ' Absences and the admin sheets visible into the roster's folder, which today is also the members' folder
    BackupBeforeSave                               ' v27 (item 12): an hourly backup when the newest is older than BACKUP_SAVE_HOURS (quiet)
    On Error GoTo 0
End Sub

Private Sub Workbook_AfterSave(ByVal Success As Boolean)
    On Error Resume Next
    If Success Then NoteSaveResult True            ' v26.5.5 (A1): a File > Save by hand clears the Desk's "Not saved since" line
    If Success Then RecordDiskStamp                ' v26.5.5 (A2): our own save is the new on-disk time
    If mRestoreAdmin Then
        mRestoreAdmin = False
        ApplyView True, mRestoreDesk               ' v25.3: the Desk view comes back as it was
        If Not mWasActive Is Nothing Then mWasActive.Activate
        Set mWasActive = Nothing
        If Success Then ThisWorkbook.Saved = True      ' v23: a failed save stays "not saved"
        Application.ScreenUpdating = True
    End If
    On Error GoTo 0
End Sub

' Closing: writable copies are saved (as before); read-only copies, or a
' workbook whose conversion failed, close without saving.
Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error Resume Next
    ReleaseStatusBar                       ' v24.1: Excel's status bar belongs to every open workbook
    SetShortcuts False                     ' v25.5: the keys go with the workbook
    If NoSave() Or ThisWorkbook.ReadOnly Then
        ThisWorkbook.Saved = True
        Exit Sub
    End If
    PublishReminderOnClose                 ' v24.1: offers to publish the members' file if it is behind
    CalHintReset                           ' v25.5: a clicked line's details are not saved for the next opener (modCalendar)
    mRestoreAdmin = True                   ' closing - no tab redraws while the view switches
    ApplyView False
    RosterSheet().Activate
    DeskZoomReset                          ' v25.3: a Desk user's 120 % is not saved for everyone
    mRestoreAdmin = False
    Dim en As Long, em As String
    Err.Clear
    ThisWorkbook.Save
    en = Err.Number: em = Err.Description
    If en <> 0 Or Not ThisWorkbook.Saved Then
        ' v26.5.5 (A1): a close that could not save is REPORTED and cancelled (the file stayed as it was on disk).
        ' BeforeSave refusing a stale copy (A2) has already said "close WITHOUT saving": that close goes through, unsaved.
        If NoSave() Then
            ThisWorkbook.Saved = True
        Else
            If em = "" Then em = "the file on disk did not change"
            LogError "Workbook_BeforeClose", en, em
            NoteSaveResult False
            Cancel = True
            ApplyView IsAdmin(), IsDeskUser()      ' the view as it was before the close started
            SetShortcuts True
            MsgBox "NOT SAVED to disk (" & em & ")." & vbNewLine & vbNewLine & _
                   "Don't close Excel. Try File > Save now; if that fails, tell " & PRIMARY_ADMIN_NAME & "." & vbNewLine & _
                   "Everything logged this session is in this workbook until it is saved.", vbCritical, "Not saved"
        End If
    End If
    On Error GoTo 0
End Sub

' ---- Calendar + Needs Attention tabs ----
' Switching to either tab redraws it only if something changed;
' double-click or right-click a line (or an empty spot) for its menu.
Private Sub Workbook_SheetActivate(ByVal Sh As Object)
    If mRestoreAdmin And Not gAdminUnlocked Then Exit Sub   ' mid-save / closing view switch
    On Error GoTo Oops
    DeskSheetActivated Sh              ' v25.3: a Desk user's zoom; the Desk refreshed when looked at
    CalSheetActivated Sh
    ReportsSheetActivated Sh           ' v24.1: the status bar goes back when you leave the roster
    Exit Sub
Oops:
    Quietly "Workbook_SheetActivate"
End Sub

' v24.1: switching to another workbook gives Excel's status bar back; v25.5: and the shortcut keys
Private Sub Workbook_Deactivate()
    On Error Resume Next                            ' never raises
    ReleaseStatusBar
    SetShortcuts False
End Sub

' v25.5: the shortcut keys are this workbook's only while it is the active one
Private Sub Workbook_Activate()
    On Error Resume Next                            ' never raises
    SetShortcuts True
End Sub

' v24.1: selecting a roster row shows that person's Refused hours by kind
' (call-out / posting) on Excel's status bar. Nothing is written to a cell.
' v25.5: a click on a Calendar line shows its details in the Calendar's hint row and on the status bar.
Private Sub Workbook_SheetSelectionChange(ByVal Sh As Object, ByVal Target As Range)
    On Error GoTo Oops
    RosterSelectionChanged Sh, Target
    CalSelectionChanged Sh, Target         ' v25.5 (modCalendar): row 3 of the Calendar only
    Exit Sub
Oops:
    Quietly "Workbook_SheetSelectionChange"
End Sub

' Any change to Absences / Coverage / Postings / Shift Switches / Holidays
' marks the calendar out of date (it redraws the next time you look at it).
' A changed unit code on the roster goes into the grade change log (v21).
' A changed crew on the roster goes into the crew history (v24).
Private Sub Workbook_SheetChange(ByVal Sh As Object, ByVal Target As Range)
    On Error GoTo Oops
    CalSheetChanged Sh
    GradeSheetChanged Sh, Target
    PeopleSheetChanged Sh, Target      ' v24: a crew typed on the roster asks for its effective date
    Exit Sub
Oops:
    ReportError "Workbook_SheetChange"
End Sub

Private Sub Workbook_SheetBeforeDoubleClick(ByVal Sh As Object, ByVal Target As Range, Cancel As Boolean)
    On Error GoTo Oops
    PeopleRosterDoubleClick Sh, Target, Cancel     ' v26.5: an admin's double-click on a roster row = Edit that person (modPeople)
    CalSheetDoubleClick Sh, Target, Cancel
    Exit Sub
Oops:
    ReportError "Workbook_SheetBeforeDoubleClick"
End Sub

Private Sub Workbook_SheetBeforeRightClick(ByVal Sh As Object, ByVal Target As Range, Cancel As Boolean)
    On Error GoTo Oops
    CalSheetRightClick Sh, Target, Cancel
    Exit Sub
Oops:
    ReportError "Workbook_SheetBeforeRightClick"
End Sub
