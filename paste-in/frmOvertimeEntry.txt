' ================================================================

Private Function Px(ByVal v As Double) As Double
    Px = v * UiScale(615)                        ' v27 (item 20, B14): per login (modUI), clamped so the 615-pt form stays on the screen
End Function

Private Function C(nm As String) As Object
    Set C = Me.Controls(nm)
End Function

Private Sub MakeLabel(nm As String, cap As String, L As Double, T As Double, W As Double, Optional ByVal pt As Double = 9.5)
    Dim o As Object
    Set o = Me.Controls.Add("Forms.Label.1", nm, True)
    o.Caption = cap: o.Left = Px(L): o.Top = Px(T)
    o.Width = Px(W): o.Height = Px(15)
    o.Font.Size = Px(pt)
End Sub

Private Sub MakeCombo(nm As String, L As Double, T As Double, W As Double)
    Dim o As Object
    Set o = Me.Controls.Add("Forms.ComboBox.1", nm, True)
    o.Left = Px(L): o.Top = Px(T): o.Width = Px(W): o.Height = Px(20): o.Style = 2

Private Sub MakeBox(nm As String, L As Double, T As Double, W As Double)

    Set o = Me.Controls.Add("Forms.TextBox.1", nm, True)

Private Sub FillCombo(nm As String, items As Variant)
    Dim i As Long
    For i = LBound(items) To UBound(items)
        C(nm).AddItem items(i)
    Next i
End Sub

' v22: a sick call opens this form already filled in (unit, date, Days / Nights);
' the crew fills from the rotation as usual, then the supervisor clicks LOAD LIST
' v25: statusTxt names who pre-filled it (the Install panel); "" keeps the sick-call wording
' v26.5.5 (B8): hrs > 0 = the hours carried from the sick call (typed, so the date / shift defaults leave them alone)
Public Sub Prefill(ByVal unit As String, ByVal d As Date, ByVal shiftLbl As String, Optional ByVal statusTxt As String = "", _
                   Optional ByVal hrs As Double = 0, Optional ByVal slotKey As String = "")
    On Error Resume Next
    mSlotKey = slotKey: mSlotDate = d: mSlotShift = shiftLbl: mSlotUnit = unit      ' v27 (item 10): the acceptance covers this line, no question
    C("cmbDept").Value = unit
    C("txtDate").Value = DateText(d)
    C("cmbDN").Value = IIf(IsNightShift(shiftLbl), "Nights", "Days")
    If hrs > 0 Then C("txtHours").Value = HoursText(hrs): mAutoHours = ""
    If statusTxt = "" Then statusTxt = "Sick call"
    C("lblStatus").Caption = TrustStrip() & vbLf & statusTxt & ": " & UnitDisplayName(unit) & " " & Format$(d, "ddd mmm d") & " " & shiftLbl & " - check the hours and click LOAD LIST." & _
                             IIf(statusTxt = "Sick call" And Not IsAdmin(), "  Wrong person? Ask " & PRIMARY_ADMIN_NAME & " to void the sick call.", "")
    On Error GoTo 0
End Sub

Private Sub AutoCrew()
    C("lblRot").ForeColor = NAVY                     ' back to normal until a DST note says otherwise
    Dim d As Date
    If Not ParseTypedDate(C("txtDate").Text, d) Then
        C("lblRot").Caption = "Pick the date and Days / Nights - the crew fills in from the rotation."
        Exit Sub
    End If
    If C("cmbDN").ListIndex < 0 Then
        C("lblRot").Caption = Format$(d, "ddd mmm d") & ":  " & RotationSummary(d) & "   - pick Days / Nights"
        Exit Sub
    End If
    Dim note As String, crew As String, night As Boolean
    night = IsNightShift(C("cmbDN").Text)
    crew = AvailableCrew(d, night, note)
    If C("cmbShift").Value = "" Then C("cmbShift").Value = CREW_ALL
    ' v21: clock-change night -> the actual hours (13 / 11) and a warning
    ' v26.5.5 (B8): the Hours box defaults on EVERY date / shift change (12, or the DST night's 13 / 11) unless the
    ' hours were typed by hand (the mAutoHours rule New Posting uses): a 3 AM call-out no longer waits on a typed 12
    Dim dst As String
    If night Then dst = DstNote(d)
    Dim want As String: want = HoursText(DefaultShiftHours(d, night, USUAL_SHIFT_HOURS))
    If Trim$(C("txtHours").Value) = "" Or (mAutoHours <> "" And Trim$(C("txtHours").Value) = mAutoHours) Then
        C("txtHours").Value = want: mAutoHours = want
    End If
    ' two lines: rotation on the first, the DST warning (or the list note) on the second
    If dst <> "" Then
        C("lblRot").ForeColor = RGB(190, 30, 30)
        C("lblRot").Caption = Format$(d, "ddd mmm d") & ":  " & RotationSummary(d) & "  ->  crews off: " & crew & _
                              IIf(note <> "", "  (" & note & ")", "") & vbLf & dst & " - check the Hours box"
    Else
        ' v26.5.5 (B1): a night is named by the evening it starts; before 7 AM the shift under way is said
        Dim hint As String: hint = ShiftHintText(d, night, Now)
        C("lblRot").Caption = Format$(d, "ddd mmm d") & ":  " & RotationSummary(d) & "  ->  crews off: " & crew & _
                              IIf(note <> "", "  (" & note & ")", "") & vbLf & _
                              IIf(hint <> "", hint & ".", "The list is everyone with the code who is free that shift.")
    End If
End Sub

Private Sub btnLoad_Click()
    If C("cmbDept").Value = "" Then MsgBox "Select a department.", vbExclamation, "Call Out Roster": Exit Sub
    If C("cmbDN").ListIndex < 0 Then MsgBox "Select Days or Nights.", vbExclamation, "Call Out Roster": Exit Sub
    If C("cmbShift").Value = "" Then MsgBox "Select a crew.", vbExclamation, "Call Out Roster": Exit Sub
    Dim dL As Date                                 ' v25: every date through the one parser
    If Not ParseTypedDate(C("txtDate").Text, dL) Then MsgBox "Enter a valid date (e.g. Oct 12).", vbExclamation, "Call Out Roster": Exit Sub

    Dim hoursStr As String: hoursStr = Trim$(C("txtHours").Value)
    If hoursStr = "" Or Not IsNumeric(hoursStr) Then MsgBox "Enter valid hours.", vbExclamation, "Call Out Roster": Exit Sub
    If CDbl(hoursStr) <= 0 Then MsgBox "Hours must be greater than zero.", vbExclamation, "Call Out Roster": Exit Sub

    If Not mKronosOk Then                     ' v20: warn if worked hours are old
        If Not KronosOkToContinue("call-out") Then Exit Sub
        mKronosOk = True
    End If

    If mCount > 0 And LoggedCount() > 0 Then
        If MsgBox("Reloading clears the " & LoggedCount() & " outcome(s) already recorded." & vbNewLine & _
                  "Reload anyway?", vbYesNo + vbExclamation, "Reload list") = vbNo Then Exit Sub
        If Not mNoDraft Then DraftClear CurrentUser()   ' (review A3) the cleared outcomes leave the draft too
    End If
    LoadListNow
End Sub

' v27 (item 17): the load itself (the validations above are the button's); False when nobody is listed
Private Function LoadListNow() As Boolean
    Dim dL As Date
    If Not ParseTypedDate(C("txtDate").Text, dL) Then Exit Function
    Dim hoursStr As String: hoursStr = Trim$(C("txtHours").Value)
    If hoursStr = "" Or Not IsNumeric(hoursStr) Then Exit Function
    Application.Calculate
    ' v24: each person's crew is the one they are on that date (crew history)
    mCount = LoadEligible(C("cmbDept").Value, CrewFilter(C("cmbShift").Value), mEntries, 999, CALLOUT_INCLUDE_DAYS, _
                          dL)
    ' only people free that shift: not scheduled, rest rule, not booked off, not on a switch
    Dim offList As String, notTxt As String
    Dim nAll As Long: nAll = mCount                     ' v27 (item 18): everyone with a code, before the free test
    If mCount > 0 Then mCount = FilterNotFree(mEntries, mCount, dL, C("cmbDN").Text, offList, False, notTxt)
    mNotListedTxt = notTxt
    mNotListed.Caption = "Not listed (" & (nAll - mCount) & ")..."
    mNotListed.Enabled = (nAll - mCount > 0)
    If notTxt <> "" Then ShowBarText "Not listed: " & Replace(notTxt, vbNewLine, "  |  ")   ' the same list on the status bar
    If mCount = 0 Then
        lstEmployees.Clear
        mCurrent = 0
        UpdateCurrentLabel                         ' review: the card and the progress line of the old list go too
        UpdateProgress
        SetOutcomeButtons False
        btnSubmit.Enabled = False
        mCallBack.Enabled = False
        mDept = C("cmbDept").Value: mShift = C("cmbShift").Value: mShiftLbl = C("cmbDN").Text: mDate = dL   ' (review LOW-12) the Not listed box names THIS load
        MsgBox "No employees found for " & C("cmbDept").Value & _
               " / " & C("cmbShift").Value & "." & IIf(notTxt <> "", vbNewLine & vbNewLine & "Not listed:" & vbNewLine & notTxt, ""), vbExclamation, "Call Out Roster"
        Exit Function
    End If

    mDept = C("cmbDept").Value: mShift = C("cmbShift").Value: mShiftLbl = C("cmbDN").Text
    mDate = dL: mHours = CDbl(hoursStr)
    mStarted = Now
    ' flag anyone already working a shift switch on this date / shift
    Dim j As Long, nSS As Long
    ReDim mSS(1 To mCount)
    ReDim mTaken(1 To mCount)
    mSeqNext = 0                                     ' v29.0.3 (review 1.1): a fresh list, nothing logged yet (LoadEligible's records carry LoggedSeq 0)
    For j = 1 To mCount
        mSS(j) = SwitchWorkingOn(mEntries(j).EmpName, mDate, mShiftLbl)
        If mSS(j) <> "" Then nSS = nSS + 1
    Next j
    mCurrent = 1
    RefreshList
    UpdateCurrentLabel
    UpdateProgress
    SetOutcomeButtons True
    btnSubmit.Enabled = False

    ' v26.5.5 (B5): old worked hours are said in red on the status line for a supervisor (an admin got the Yes / No)
    Dim kLine As String: kLine = KronosStaleLine()
    C("lblStatus").ForeColor = IIf(kLine <> "", RGB(190, 30, 30), RGB(51, 51, 51))
    C("lblStatus").Caption = TrustStrip() & vbLf & IIf(kLine <> "", kLine & "   |   ", "") & _
                             C("cmbDept").Value & " | " & mShiftLbl & " | Crew " & C("cmbShift").Value & _
                             " | " & C("txtDate").Value & " | " & hoursStr & " hrs" & _
                             IIf(nSS > 0, "   |   " & nSS & " working a shift switch (S/S)", "") & _
                             IIf(offList <> "", "   |   Not listed (booked off): " & offList, "")
    ' v27 (item 25, 1.4): the unit's call-out note from the Min Staff sheet, in the rotation line's place once the list is
    ' loaded (the line holds two lines and the crew is picked by now); a red DST warning keeps the line (re-review M2);
    ' an earlier load's note is put back to its rotation line first
    If Left$(C("lblRot").Caption, 11) = "Unit note (" Then
        C("lblRot").Caption = mRotTxt: C("lblRot").ForeColor = NAVY
    End If
    Dim unitNote As String: unitNote = MinStaffNote(mDept, mShiftLbl)
    If unitNote <> "" And C("lblRot").ForeColor <> RGB(190, 30, 30) Then
        mRotTxt = C("lblRot").Caption
        C("lblRot").ForeColor = RGB(150, 75, 0)
        C("lblRot").Caption = "Unit note (" & UnitDisplayName(mDept) & " " & mShiftLbl & "): " & unitNote
    End If
    FocusList
    LoadListNow = True
End Function


' ================================================================
' LIST DISPLAY
' ================================================================
Private Sub RefreshList()
    lstEmployees.Clear
    Dim i As Long, displayName As String, outcomeText As String
    ' v24.1: the hours each Accepted row will carry (they depend on partial acceptances logged before it)
    Dim oc() As String, offers() As Double, accH As Double, sq() As Long
    If mCount > 0 Then
        OutcomeArrays oc, sq
        CalloutOffers mHours, oc, mTaken, mCount, offers, sq
    End If
    For i = 1 To mCount
        If mEntries(i).Grade > CHARGE_MAX_GRADE Then
            displayName = ">> " & mEntries(i).EmpName
        Else
            displayName = " " & mEntries(i).EmpName
        End If

        If mEntries(i).Outcome = "" Then
            outcomeText = "-"
        ElseIf mEntries(i).Outcome = "Unavailable" Then
            outcomeText = "Unavail: " & mEntries(i).Reason
        ElseIf mEntries(i).Outcome = "Accepted" Then
            accH = AcceptedHours(offers(i), mTaken(i))

        End If

        If mSS(i) <> "" Then displayName = displayName & "  [S/S]"

' v25.5: "4 of 12 logged  -  2 left messages  -  3 to call  -  started 14 min ago" (no timer: refreshed on every outcome)
Private Sub UpdateProgress()

    Dim i As Long, logged As Long, nMsg As Long
    For i = 1 To mCount
        If mEntries(i).Outcome <> "" Then logged = logged + 1
        If mEntries(i).Outcome = "Left Message" Then nMsg = nMsg + 1
    Next i
    Dim txt As String
    If logged = mCount Then
        txt = "All " & mCount & " logged - Enter submits"
    ElseIf logged = 0 Then
        txt = "0 of " & mCount & " logged - a key or a button starts"
    Else
        txt = logged & " of " & mCount & " logged"
    End If

    txt = txt & "  -  " & IIf(mins < 1, "just started", "started " & mins & " min ago")
    C("lblProgress").Caption = txt
    mCallBack.Enabled = (nMsg > 0) And Not SessionClosed()   ' v25.5 review: no call-backs once a partial closed the shift
End Sub

Private Sub lstEmployees_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If Shift <> 0 Then Exit Sub                      ' Ctrl / Alt / Shift combinations are not ours
    If mKeyHeld <> 0 And KeyCode = mKeyHeld Then KeyCode.Value = 0: Exit Sub   ' a repeat of the key still down
    Dim handled As Boolean: handled = True
    Select Case KeyCode
        Case vbKeyA, vbKeyR, vbKeyL, vbKeyU, vbKeyP, vbKeyK, vbKeyB, vbKeyReturn, vbKeyEscape: mKeyHeld = KeyCode   ' ours: one press, one outcome (re-review: arrows and Tab keep repeating)
    End Select

        Case vbKeyA: If btnAccepted.Enabled Then btnAccepted_Click
        Case vbKeyR: If btnRefused.Enabled Then btnRefused_Click

        Case vbKeyEscape: KeyCode.Value = 0: Unload Me: Exit Sub     ' the unsaved-entries question as always (QueryClose); nothing after the unload (review)
        Case Else: handled = False
    End Select
    If handled Then KeyCode.Value = 0
End Sub

' Hours on offer to person idx: the call-out's hours, or 0 once a partial acceptance logged before them closed the shift
Private Function OfferFor(ByVal idx As Long) As Double
    Dim oc() As String, offers() As Double, sq() As Long
    OfferFor = mHours
    If mCount < 1 Or idx < 1 Or idx > mCount Then Exit Function
    OutcomeArrays oc, sq
    CalloutOffers mHours, oc, mTaken, mCount, offers, sq
    OfferFor = offers(idx)
End Function

' v29.0.3 (review 1.1): every row's outcome and the order it was logged in, as CalloutOffers reads them
Private Sub OutcomeArrays(ByRef oc() As String, ByRef sq() As Long)
    Dim i As Long
    ReDim oc(1 To mCount): ReDim sq(1 To mCount)
    For i = 1 To mCount
        oc(i) = mEntries(i).Outcome
        sq(i) = mEntries(i).LoggedSeq
        If oc(i) = "" Then sq(i) = mSeqNext + 1      ' (re-review R1) a row not yet logged comes AFTER everything logged, so the card reads "offering 0" for every uncalled row once the shift closed
    Next i
End Sub

' v25 (D-E4): once someone took PART of the hours the shift is closed - no further outcome of any
' kind (v29.0.3: a row logged after the close is offered 0 by its logged order, so it could never be
' charged; the refusal is still refused here so the list is not called on). The partial taker's own
' row stays correctable (their entry is left out of the test).
' v25.5 (review): a row logged BEFORE the close may still be corrected to a refusal, a message or
' unavailable - never to Accepted (toAccepted): Call back and B reach those rows with one key now.
Private Function ShiftClosedByPartial(Optional ByVal toAccepted As Boolean = False) As Boolean
    Dim oc() As String, tk() As Double, i As Long
    If mCount < 1 Then Exit Function
    If mCurrent >= 1 And mCurrent <= mCount Then
        If mEntries(mCurrent).Outcome <> "" And Not toAccepted Then Exit Function   ' a row already logged (before the close) can be corrected, not to Accepted
    End If
    ReDim oc(1 To mCount): ReDim tk(1 To mCount)
    For i = 1 To mCount
        If i <> mCurrent Then oc(i) = mEntries(i).Outcome: tk(i) = mTaken(i)
    Next i
    If CalloutClosed(oc, tk, mCount, mHours) Then
        ShiftClosedByPartial = True
        MsgBox "A partial acceptance closed this shift: the remaining hours are dropped, uncharged, and nobody else is called." & vbNewLine & vbNewLine & _
               "Leave the rest of the list uncalled and click SUBMIT ALL ENTRIES.", vbExclamation, "Shift closed"
    End If
End Function

' P / ACCEPTED PART OF THE HOURS: asks how many, records Accepted with those hours
Private Sub mPartAcc_Click()
    If mCurrent < 1 Or mCurrent > mCount Then Exit Sub
    If ShiftClosedByPartial(True) Then Exit Sub
    Dim was As Double: was = mTaken(mCurrent)
    mTaken(mCurrent) = 0
    Dim offer As Double: offer = OfferFor(mCurrent)
    Dim a As Variant, h As Double
    a = Application.InputBox(Prompt:="Hours " & mEntries(mCurrent).EmpName & " is taking." & vbNewLine & vbNewLine & _
                             HoursText(offer) & " h are on offer. Enter fewer hours: the rest is dropped, uncharged, and the shift closes.", _
                             Title:="Accepted part of the hours", Type:=1)
    If VarType(a) = vbBoolean Then mTaken(mCurrent) = was: FocusList: Exit Sub
    h = CDbl(a)
    If h <= 0 Then mTaken(mCurrent) = was: MsgBox "The hours must be more than 0.", vbExclamation, "Call Out Roster": FocusList: Exit Sub
    If h >= offer Then h = 0                              ' the whole of what was on offer
    If IIf(h > 0, h, offer) > mEntries(mCurrent).MaxHours + 0.001 Then      ' v26.5.5 (item 9)
        mTaken(mCurrent) = was
        MsgBox FatigueRefusalText(mCurrent), vbExclamation, "16 h rule": FocusList: Exit Sub
    End If
    mTaken(mCurrent) = h
    If Not RecordOutcome("Accepted", "", True) Then mTaken(mCurrent) = was     ' (review M4) the cap box cancelled: as it was
End Sub

                         "OT accepted, an award or a switch touches it). Press P and enter the hours they are taking, or log the outcome they gave."
End Function

' v27 (item 21, review M4): a Function - False = the weekly-cap reason box was cancelled and NOTHING changed (the
' partial-hours caller puts its value back)
Private Function RecordOutcome(ByVal outcome As String, ByVal reason As String, Optional ByVal keepTaken As Boolean = False) As Boolean
    If mCurrent < 1 Or mCurrent > mCount Then Exit Function
    If outcome = "Accepted" And Not keepTaken Then       ' v26.5.5 (item 9): the whole offer must fit the 16 h rule
        If mEntries(mCurrent).MaxHours < OfferFor(mCurrent) - 0.001 Then
            MsgBox FatigueRefusalText(mCurrent), vbExclamation, "16 h rule": FocusList: Exit Function
        End If
    End If
    Dim takenWas As Double: takenWas = mTaken(mCurrent)
    If Not keepTaken Then mTaken(mCurrent) = 0           ' only ACCEPTED PART sets partial hours
    ' v27 (item 21): the weekly cap - a warning and a reason into the row, never a block; a Cancel leaves the row as it was
    If outcome = "Accepted" Then
        Dim wc As String
        wc = WeekCapProblem(mEntries(mCurrent).EmpName, mEntries(mCurrent).EmpShift, mDate, mShiftLbl, IIf(keepTaken And mTaken(mCurrent) > 0, mTaken(mCurrent), OfferFor(mCurrent)))
        If wc <> "" Then
            MsgBox wc & vbNewLine & vbNewLine & "The weekly overtime limit is " & HoursText(WeekCap()) & " h. It can be accepted with a reason, which goes into the log row (Cancel = the acceptance is not logged).", vbExclamation, "Weekly hours"
            Dim wr As String: wr = AskRequired("Why is this acceptance OK past the weekly limit?", "Weekly hours")
            If wr = "" Then mTaken(mCurrent) = takenWas: FocusList: Exit Function
            reason = Trim$(reason & " [over " & HoursText(WeekCap()) & " h/week: " & wr & "]")
        End If
    End If
    RecordOutcome = True

    ' v29.0.3 (review 1.1): the order outcomes are LOGGED in, for the partial-acceptance close (CalloutOffers) - a NEW number
    ' for an Accepted (a call-back who takes part of the hours closes the shift from THAT moment, after the refusals already
    ' logged) or a row logged for the first time; a correction of a row logged before keeps its place
    If outcome = "Accepted" Or mEntries(mCurrent).LoggedSeq = 0 Then mSeqNext = mSeqNext + 1: mEntries(mCurrent).LoggedSeq = mSeqNext
    mEntries(mCurrent).Outcome = outcome
    mEntries(mCurrent).Reason = reason
    C("cmbReason").Value = ""      ' so it can't carry over to the next person

    Dim i As Long, nextIdx As Long, logged As Long
    logged = LoggedCount()
    For i = mCurrent + 1 To mCount
        If mEntries(i).Outcome = "" Then nextIdx = i: Exit For
    Next i
    If nextIdx = 0 Then
        For i = 1 To mCount
            If mEntries(i).Outcome = "" Then nextIdx = i: Exit For
        Next i
    End If

    If nextIdx > 0 Then
        mCurrent = nextIdx
    Else
        mCurrent = mCount + 1
    End If

    RefreshList
    UpdateCurrentLabel
    UpdateProgress
    btnSubmit.Enabled = (logged > 0)
    FocusList
    If Not mResuming Then DraftSave                      ' v27 (item 17): the session so far, for a resume after a crash
End Function

' v27 (item 17, B4): this session's entries to the Draft sheet (quiet; a draft that cannot be written is simply not there)
Private Function DraftSave() As Boolean
    If mCount = 0 Or mNoDraft Then Exit Function
    ' (review A2 / MED-1) a stored draft that is NOT this session's (kept for later, or left by a crash) is never replaced
    ' without the question - asked once, on the first write, by what the table holds, however this session started
    If Not mDraftAsked And Not mResumedDraft Then
        mDraftAsked = True
        Dim kd As String, ks As String, kc As String, ke As String, ko As String, kdt As Date, kh As Double, kst As Date
        If DraftRead(CurrentUser(), kd, ks, kc, kdt, kh, kst, ke, ko) Then
            If MsgBox("A draft of yours is stored: " & kd & " " & ks & " " & Format$(kdt, "ddd mmm d") & " (" & (UBound(Split(ko, ";")) + 1) & " logged)." & vbNewLine & vbNewLine & _
                      "Replace it with THIS session's draft?  (No = keep the old one; this session is not drafted.)", vbYesNo + vbQuestion + vbDefaultButton2, "Call-out session") = vbNo Then
                mNoDraft = True
                Exit Function
            End If
        End If
    End If
    Dim i As Long, s As String
    For i = 1 To mCount
        If mEntries(i).Outcome <> "" Then
            s = s & IIf(s = "", "", ";") & Replace(Replace(mEntries(i).EmpName, "|", " "), ";", " ") & "|" & mEntries(i).Outcome & "|" & _
                Replace(Replace(mEntries(i).Reason, "|", " "), ";", " ") & "|" & HoursText(mTaken(i)) & "|" & CStr(mEntries(i).LoggedSeq)   ' v29.0.3: 5th = the logged order
        End If
    Next i
    Dim slotKey As String
    If mSlotKey <> "" And Int(mDate) = Int(mSlotDate) And StrComp(mShiftLbl, mSlotShift, vbTextCompare) = 0 _
       And StrComp(mDept, mSlotUnit, vbTextCompare) = 0 Then slotKey = mSlotKey
    DraftSave = DraftWrite(CurrentUser(), mDept, mShiftLbl, mShift, mDate, mHours, mStarted, slotKey, s)
    ' (review A1) the row is in the workbook; the file is saved every DRAFT_SAVE_MINUTES so a crash loses minutes, not the session
    If DraftSave And Not gSmokeRun Then
        If (Now - mDraftSavedAt) * 1440 >= DRAFT_SAVE_MINUTES Then
            SaveIfWritable                           ' (review MED-3) the clock restarts whatever the result: a failed save is not retried at every outcome
            mDraftSavedAt = Now
        End If
    End If
End Function

' v27 (item 17): shown fresh (no Prefill, nothing loaded) by a login with a draft: offer to pick it up
Private Sub UserForm_Activate()
    If mActivated Then Exit Sub
    mActivated = True
    If mCount > 0 Then Exit Sub
    If C("cmbDept").Value <> "" Or mSlotKey <> "" Then Exit Sub     ' a sick call's or a calendar line's call-out is a new session
    On Error GoTo Quiet
    Dim dept As String, sh As String, crewSel As String, d As Date, hrs As Double, startedAt As Date, slotKey As String, outcomes As String
    If Not DraftRead(CurrentUser(), dept, sh, crewSel, d, hrs, startedAt, slotKey, outcomes) Then Exit Sub
    Dim nOut As Long: nOut = UBound(Split(outcomes, ";")) + 1
    If outcomes = "" Then nOut = 0
    Dim ans As Long
    ans = MsgBox("Resume the " & dept & " " & sh & " " & Format$(d, "ddd mmm d") & " session (" & nOut & " logged, started " & _
                 Format$(startedAt, "h:mm AM/PM") & IIf(Int(startedAt) <> Int(Now), " " & Format$(startedAt, "ddd"), "") & ")?" & vbNewLine & vbNewLine & _
                 "Yes = load that list again and put the outcomes back.  No = discard it.  Cancel = leave it for later.", _
                 vbYesNoCancel + vbQuestion, "Call-out session")
    If ans = vbNo Then DraftClear CurrentUser(): Exit Sub
    If ans <> vbYes Then Exit Sub
    ' (review A6) a unit or crew the lists no longer offer is said, not swallowed; the draft stays for No
    If Not ComboHas(C("cmbDept"), dept) Then
        MsgBox "The draft's unit (" & dept & ") is not in the unit list any more. Discard the draft with No next time, or ask " & PRIMARY_ADMIN_NAME & ".", vbExclamation, "Call-out session"
        Exit Sub
    End If
    ' the inputs as the draft had them, then the list, then the outcomes by name
    C("cmbDept").Value = dept
    C("cmbDN").Value = IIf(IsNightShift(sh), "Nights", "Days")
    C("txtDate").Value = DateText(d)
    C("txtHours").Value = HoursText(hrs): mAutoHours = ""
    If crewSel <> "" Then
        If ComboHas(C("cmbShift"), crewSel) Then C("cmbShift").Value = crewSel Else C("cmbShift").Value = CREW_ALL
    End If
    mSlotKey = slotKey: mSlotDate = d: mSlotShift = sh: mSlotUnit = dept
    mResuming = True
    If Not LoadListNow() Then mResuming = False: Exit Sub
    mResumedDraft = True                             ' this session's draft from here on (review: only once the list loaded)
    If startedAt > 0 Then mStarted = startedAt
    Dim parts As Variant, p As Variant, i As Long, k As Long, placed As Long, missing As String
    parts = Split(outcomes, ";")
    For k = 0 To UBound(parts)
        p = Split(parts(k), "|")
        If UBound(p) >= 1 Then
            For i = 1 To mCount
                If StrComp(Trim$(mEntries(i).EmpName), Trim$(CStr(p(0))), vbTextCompare) = 0 Then Exit For
            Next i
            If i <= mCount Then
                mEntries(i).Outcome = CStr(p(1))
                If UBound(p) >= 2 Then mEntries(i).Reason = CStr(p(2))
                If UBound(p) >= 3 Then mTaken(i) = Val(Replace(CStr(p(3)), ",", "."))
                ' v29.0.3 (review 1.1): the logged order comes back with the row; a draft from before it numbers its rows in draft order
                If UBound(p) >= 4 Then mEntries(i).LoggedSeq = CLng(Val(CStr(p(4)))) Else mEntries(i).LoggedSeq = k + 1
                If mEntries(i).LoggedSeq > mSeqNext Then mSeqNext = mEntries(i).LoggedSeq
                placed = placed + 1
            Else
                missing = missing & IIf(missing = "", "", ", ") & CStr(p(0))
            End If
        End If
    Next k
    mResuming = False
    ' the card on the first person still without an outcome
    mCurrent = mCount + 1
    For i = 1 To mCount
        If mEntries(i).Outcome = "" Then mCurrent = i: Exit For
    Next i
    RefreshList
    UpdateCurrentLabel
    UpdateProgress
    btnSubmit.Enabled = (LoggedCount() > 0)
    C("lblStatus").Caption = TrustStrip() & vbLf & "Resumed: " & placed & " of " & nOut & " outcome(s) put back" & _
                             IIf(missing <> "", " - not on this list any more: " & missing, "") & ".  Carry on, then SUBMIT."
    If missing <> "" Then C("lblStatus").ForeColor = RGB(190, 30, 30)
    FocusList
    Exit Sub
Quiet:
    mResuming = False
    LogError "frmOvertimeEntry.Resume", Err.Number, Err.Description
End Sub

' True when the combo's list holds the text (case-insensitive)
Private Function ComboHas(ByVal cb As Object, ByVal txt As String) As Boolean
    On Error Resume Next                            ' never raises
    Dim i As Long
    For i = 0 To cb.ListCount - 1
        If StrComp(Trim$(CStr(cb.List(i, 0))), Trim$(txt), vbTextCompare) = 0 Then ComboHas = True: Exit Function
    Next i
End Function


' ================================================================
' SUBMIT
' Charged: Refused + Left Message, grade <= CHARGE_MAX_GRADE,
'          at CHARGE_MULTIPLIER, to the pay period of the callout date.
' Not charged: Accepted, Unavailable, anyone above the grade limit.
' ================================================================
Private Sub btnSubmit_Click()
    Dim logged As Long: logged = LoggedCount()
    If logged = 0 Then MsgBox "No outcomes recorded yet.", vbExclamation, "Call Out Roster": Exit Sub

    If mCount - logged > 0 Then
        If MsgBox(mCount - logged & " employee(s) have no outcome." & vbNewLine & _
                  "Submit the " & logged & " completed entries?", _
                  vbYesNo + vbQuestion, "Incomplete Session") = vbNo Then Exit Sub
    End If

    If InputsChanged() Then
        If MsgBox("The department, shift, crew, date or hours were changed after LOAD LIST." & vbNewLine & vbNewLine & _
                  "Submit using the values the list was loaded with?" & vbNewLine & _
                  "  " & mDept & " | " & mShiftLbl & " | " & mShift & " | " & Format$(mDate, "ddd mmm d") & " | " & mHours & " hrs" & _
                  vbNewLine & vbNewLine & "(No = cancel; click LOAD LIST again to use the new values.)", _
                  vbYesNo + vbExclamation, "Inputs changed") = vbNo Then Exit Sub
    End If
    ' v24.2: the rows are built and written by ONE function in modOvertimeEntry (SmokeTest uses it too)
    ' v25.5: the box shows the four-line summary; the full detail goes to the status bar (until the next click) and is in the OT Log
    Dim sessionID As String, summary As String, shortSummary As String
    Dim slotKey As String                           ' v27 (item 10): the line the call-out was started from, if the date and shift still match
    If mSlotKey <> "" And Int(mDate) = Int(mSlotDate) And StrComp(mShiftLbl, mSlotShift, vbTextCompare) = 0 _
       And StrComp(mDept, mSlotUnit, vbTextCompare) = 0 Then slotKey = mSlotKey     ' (review M6) a changed unit drops it too
    If Not SubmitCalloutSession(mDept, mShiftLbl, mShift, mDate, mHours, mEntries, mTaken, mCount, sessionID, summary, shortSummary, slotKey, mNoDraft) Then Exit Sub   ' form keeps its entries so nothing is lost; (MED-2) a kept draft is not this session's to clear
    ShowBarText Replace(summary, vbNewLine, "  |  ")
    If gSaveFailedAt > 0 Or NoSave() Then
        ' v26.5.5 (A1): the rows are in the OT Log but the file did not save (or saving is blocked for the session).
        ' The entries stay on screen so nothing is lost from view; SUBMIT is locked so they can't be logged twice.
        ' The pop-up is modal, so the way out is: close it, then File > Save (or close the roster without saving
        ' and log the session again, when saving is blocked) - SaveBlockText says which.
        mSaveFailed = True
        MsgBox shortSummary, vbCritical, "Session " & sessionID & " logged - NOT SAVED"
        btnSubmit.Enabled = False
        SetOutcomeButtons False
        C("lblStatus").ForeColor = RGB(190, 30, 30)
        C("lblStatus").Caption = SaveBlockText() & "  (Session " & sessionID & ")"
        MarkChangedTotals                         ' v27 (item 8, review M2): the rows ARE in the log - the marks go on under this session's ref, not a later card's
        Exit Sub
    End If
    ShowReceipt                                   ' v27 (item 7): the receipt card (SubmitCalloutSession filled it); the detail stays on the status bar

    ResetForm
End Sub

Private Function InputsChanged() As Boolean
    Dim d As Variant, h As Variant
    d = C("txtDate").Value: h = Trim$(C("txtHours").Value)
    If C("cmbDept").Value <> mDept Or C("cmbShift").Value <> mShift Or C("cmbDN").Text <> mShiftLbl Then InputsChanged = True: Exit Function
    Dim dI As Date
    If Not ParseTypedDate(CStr(d), dI) Or Not IsNumeric(h) Or h = "" Then InputsChanged = True: Exit Function
    If dI <> mDate Or CDbl(h) <> mHours Then InputsChanged = True
End Function

Private Sub ResetForm()
    mCount = 0: mCurrent = 0
    mNotListedTxt = "": mNotListed.Caption = "Not listed...": mNotListed.Enabled = False   ' v27 (item 18)
    mSlotKey = "": mSlotShift = "": mSlotUnit = ""   ' v27 (item 10)
    mDraftAsked = False: mResumedDraft = False: mNoDraft = False   ' v27 (item 17): the next session asks afresh
    mStarted = 0
    lstEmployees.Clear
    C("cmbDept").Value = "": C("cmbShift").Value = CREW_ALL: C("cmbDN").Value = ""
    C("cmbReason").Value = ""
    C("txtHours").Value = "": mAutoHours = ""
    C("txtDate").Value = DateText(DefaultShiftDate(Now))       ' v26.5.5 (B1)
    UpdateCurrentLabel
    UpdateProgress
    C("lblStatus").Caption = TrustStrip()            ' v26.5.5 (5.2)
    C("lblStatus").ForeColor = RGB(51, 51, 51)
    SetOutcomeButtons False
    btnSubmit.Enabled = False
End Sub


' ================================================================
' WARN ON CLOSE (the old version unloaded even when you said No)
' ================================================================
Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    RestartMarkClock                               ' v27 (item 8, re-review L5): the roster's marks were painted under this form - their 60 s run from here
    If gErrorClosing Then Exit Sub                 ' v24.2: closed by the error handler, which says what happened
    If mSaveFailed Then Exit Sub                   ' v26.5.5: the entries are logged already; closing is the way to File > Save
    If mCount = 0 Then Exit Sub
    Dim logged As Long: logged = LoggedCount()
    If logged > 0 Then
        ' v27 (item 17): the entries are in the draft - keep it for a resume, discard it, or stay. (MED-2) a session that is
        ' NOT drafted (the older draft was kept) says so: closing loses its entries, and the older draft is left alone
        If mNoDraft Then
            If MsgBox(logged & " entries are not submitted and NOT drafted (you kept the older draft)." & vbNewLine & "Close and lose them?", _
                      vbYesNo + vbExclamation + vbDefaultButton2, "Unsaved entries") = vbNo Then Cancel = 1
            Exit Sub
        End If
        Select Case MsgBox(logged & " entries are not submitted. They are kept as a draft and offered the next time you open Fill a shift now." & vbNewLine & vbNewLine & _
                           "Yes = close and keep the draft.   No = close and DISCARD the entries.   Cancel = stay here.", _
                           vbYesNoCancel + vbExclamation, "Unsaved entries")
            Case vbYes
                If Not DraftSave() Then MsgBox "The draft could not be written (the Draft sheet): these entries are NOT kept.", vbExclamation, "Unsaved entries"
            Case vbNo: DraftClear CurrentUser()
            Case Else: Cancel = 1
        End Select
    End If
End Sub
