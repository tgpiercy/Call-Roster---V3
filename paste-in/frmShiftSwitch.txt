' ================================================================
' SHIFT SWITCH (S/S)
' One person works another's shift on one date. The bottom list shows
' switches still OWED a payback. Click one and the form fills in the
' reverse switch (people swapped, same unit, "Payback for" set) - just
' enter the payback date and save. Once paid back it leaves the list.
' Two-way swap: tick it and enter the payback date to log both halves
' at once (also for a same-day days <-> nights swap between sister crews).
' Hard stops: nobody booked off (vac / training / leave) or already on a
' switch that day. Shift-rotation checks are warnings only.
'
' Every control is built in code below. Anything drawn on the form in
' the designer is hidden automatically.
'
' v20 (Oct 9, 2026):
'  - A date typed without a year ("Jan 5") that is more than 90 days ago
'    means next year. Dates more than ~13 months away read as unfinished.
'  - Lists only rebuild when the date actually changes.
'  - Changing the unit clears the two names picked for the old unit.
'  - Two-way swap no longer shows a pre-picked "Payback for".
'  - The payback hint shows booked off / already on a switch before Save.
'  - Voiding an owed switch clears the form it filled in.
'  - Enter no longer saves; click SAVE SWITCH.
'  - Uses PersonFlag's kind code instead of matching its wording.
'  - DAYS workers: the shift fills in, and stat holidays count as off.
' ================================================================

Private Sub FillCombo(ByVal cb As Object, ByVal items As Variant)
    Dim i As Long
    cb.Clear
    If Not IsArray(items) Then Exit Sub
    For i = LBound(items) To UBound(items)
        cb.AddItem CStr(items(i))
    Next i
End Sub

' ================================================================
' LIVE CHECKS
' ================================================================
Private Function PickedDate(ByRef d As Date) As Boolean
    PickedDate = ParseShiftDate(mDate.Text, d)
End Function

' v25: the one parser (modPeople.ParseTypedDate); this wrapper keeps the form's call sites
Private Function ParseShiftDate(ByVal t As String, ByRef d As Date) As Boolean
    ParseShiftDate = ParseTypedDate(t, d)
End Function

Private Sub mDate_Change()
    Dim d As Date, ok As Boolean, key As String
    ok = PickedDate(d)
    TypedDateHint mDate.Text, C("hDate")
    If Trim$(mDate.Text) = "" Then C("hDate").Caption = "e.g. Oct 12 or 10/12/2026"
    If ok Then key = Format$(d, "yyyy-mm-dd")
    If key = mLastDateKey Then Exit Sub   ' same date as before (or still unfinished) - nothing to rebuild
    mLastDateKey = key
    RefillPeople
    Refresh True
End Sub

Private Sub mUnit_Change()
    Dim wasFilling As Boolean: wasFilling = mFilling
    If Not wasFilling Then
        ' names picked for the old unit don't carry over
        mFilling = True
        mOff.ListIndex = -1: mWorks.ListIndex = -1
        mFilling = False
    End If
    RefillPeople
    Refresh True
    If Not wasFilling Then LoadPayback
End Sub

Private Sub mOff_Change()
    If mRefilling Then Exit Sub
    Refresh True
    RefillWorks                          ' the shift comes from this person - re-check who's free
    If Not mFilling Then LoadPayback
End Sub

Private Sub mWorks_Change()
    If mRefilling Then Exit Sub
    Refresh False
    If Not mFilling Then LoadPayback
End Sub

Private Sub mShift_Change()
    If Not mAutoShift Then Refresh False
    RefillWorks
End Sub

' ---- name lists: only people signed off in the unit and on / free for the shift ----
' Gives up: signed off in the unit and scheduled that date. Needs the date and unit.
Private Sub RefillPeople()
    Dim unit As String: unit = mUnit.Text
    Dim d As Date, ok As Boolean
    ok = PickedDate(d) And unit <> ""
    mRefilling = True
    If ok Then
        FillPeople mOff, SwitchPickList(unit, d, "OFF", "", False)
    Else
        FillPeople mOff, Empty
    End If
    mRefilling = False
    RefillWorks
    If Not ok Then
        C("status").Caption = "Enter the shift date and unit first - the lists fill with who's scheduled and who's free."
    ElseIf Left$(C("status").Caption, 9) = "Enter the" Or Left$(C("status").Caption, 4) = "Pick" Then
        C("status").Caption = ""
    End If
End Sub

' Works: signed off in the unit and free for that shift - filled once the person
' giving it up (or the shift) is picked; never lists the person giving it up.
Private Sub RefillWorks()
    Dim d As Date, sh As String, items As Variant, i As Long, k As Long, outA() As String
    If mShift.ListIndex >= 0 Then sh = mShift.Text
    If PickedDate(d) And mUnit.Text <> "" And (mOff.ListIndex >= 0 Or sh <> "") Then
        items = SwitchPickList(mUnit.Text, d, "WORKS", sh, SameDaySwap())
        If IsArray(items) And mOff.ListIndex >= 0 Then
            ReDim outA(0 To UBound(items) - LBound(items))
            For i = LBound(items) To UBound(items)
                If StrComp(CStr(items(i)), mOff.Text, vbTextCompare) <> 0 Then outA(k) = CStr(items(i)): k = k + 1
            Next i
            If k = 0 Then
                items = Empty
            Else
                ReDim Preserve outA(0 To k - 1)
                items = outA
            End If
        End If
    End If
    mRefilling = True
    FillPeople mWorks, items
    mRefilling = False
End Sub

' fills a name list; a name already picked stays (its hint line shows any problem)
Private Sub FillPeople(ByVal cb As Object, ByVal items As Variant)
    Dim keepName As String, i As Long, found As Boolean
    If cb.ListIndex >= 0 Then keepName = cb.Text
    cb.Clear
    If IsArray(items) Then
        For i = LBound(items) To UBound(items)
            cb.AddItem CStr(items(i))
            If StrComp(CStr(items(i)), keepName, vbTextCompare) = 0 Then found = True
        Next i
    End If
    If keepName <> "" Then
        If Not found Then cb.AddItem keepName, 0
        cb.Value = keepName
    End If
End Sub

Private Sub Refresh(ByVal setShift As Boolean)
    Dim d As Date
    If Not PickedDate(d) Then
        C("hOff").Caption = "": C("hWorks").Caption = "": C("hShift").Caption = ""
        Exit Sub
    End If

    ' shift: from the person giving it up
    If setShift And mOff.ListIndex >= 0 Then
        Dim sch As String: sch = ScheduledShift(mOff.Text, d)
        If sch <> "" Then
            mAutoShift = True
            mShift.Value = sch
            mAutoShift = False
            C("hShift").Caption = "from " & ShortName(mOff.Text) & "'s crew"
        Else
            C("hShift").Caption = ""
        End If
    End If

    Dim isNight As Boolean: isNight = (mShift.Text = "Nights")
    Dim msg As String
    Dim flag As String
    If mOff.ListIndex >= 0 Then
        msg = CrewStatus(mOff.Text, d)
        flag = PersonFlag(mOff.Text, d, , mShift.Text)
        Dim oc As String
        If mShift.ListIndex >= 0 Then oc = OffConflict(mOff.Text, d, isNight)
        If flag <> "" Then
            msg = msg & "  -  NOT ELIGIBLE: " & flag
        ElseIf oc <> "" Then
            msg = msg & "  -  check: " & oc
        End If
        SetHint "hOff", msg, (oc = "" And flag = "")
    Else
        C("hOff").Caption = ""
    End If

    If mWorks.ListIndex >= 0 Then
        msg = CrewStatus(mWorks.Text, d)
        Dim wkKind As String
        flag = PersonFlag(mWorks.Text, d, wkKind, mShift.Text)
        If wkKind = "GAVE_UP" Then flag = ""     ' gave up their own shift that day - fine for the one working
        Dim wc As String
        If mShift.ListIndex >= 0 And Not SameDaySwap() Then wc = WorksConflict(mWorks.Text, d, isNight)
        If flag <> "" Then
            msg = msg & "  -  NOT ELIGIBLE: " & flag
        ElseIf wc <> "" Then
            msg = msg & "  -  check: " & wc
        End If
        SetHint "hWorks", msg, (wc = "" And flag = "")
    Else
        C("hWorks").Caption = ""
    End If
    PayHint
End Sub

' ---- two-way swap ----
Private Function PayDateOk(ByRef d2 As Date) As Boolean
    If Not mTwoWay.Value Then Exit Function
    PayDateOk = ParseShiftDate(mPayDate.Text, d2)
End Function

' same-date days <-> nights swap between sister crews
Private Function SameDaySwap() As Boolean
    Dim d As Date, d2 As Date
    If PickedDate(d) And PayDateOk(d2) Then SameDaySwap = (Int(d) = Int(d2))
End Function

' payback shift = the shift the person who works today is scheduled for on the payback date
Private Function PayShift(ByVal d2 As Date) As String
    If mWorks.ListIndex < 0 Then Exit Function
    PayShift = ScheduledShift(mWorks.Text, d2)
End Function

Private Sub PayHint()
    Dim d2 As Date
    If Not mTwoWay.Value Then C("hPay").Caption = "": Exit Sub
    If Trim$(mPayDate.Text) = "" Then C("hPay").Caption = "payback date": C("hPay").ForeColor = RGB(120, 120, 120): Exit Sub
    If Not PayDateOk(d2) Then SetHint "hPay", "not a date", False: Exit Sub
    If mOff.ListIndex < 0 Or mWorks.ListIndex < 0 Then SetHint "hPay", Format$(d2, "ddd mmm d"), True: Exit Sub
    Dim sh As String: sh = PayShift(d2)
    Dim prob As String: prob = PaybackProblem(d2, sh)
    If prob = "" Then
        ' the same hard stops Save checks for the payback half (booked off / already on a switch)
        Dim d As Date
        If PickedDate(d) Then
            If Int(d2) <> Int(d) Then prob = FirstProblem(EligibilityProblems(mWorks.Text, mOff.Text, d2, sh))
        End If
    End If
    If prob <> "" Then
        SetHint "hPay", Format$(d2, "ddd mmm d") & ":  NOT POSSIBLE - " & prob, False
    Else
        SetHint "hPay", Format$(d2, "ddd mmm d") & " " & sh & ": " & ShortName(mOff.Text) & " works for " & ShortName(mWorks.Text), True
    End If
End Sub

' v25: the rule is modShiftSwitch.PaybackProblem; this wrapper keeps the form's call sites
Private Function PaybackProblem(ByVal d2 As Date, ByVal sh2 As String) As String
    If mOff.ListIndex < 0 Or mWorks.ListIndex < 0 Then Exit Function
    Dim d As Date
    If Not PickedDate(d) Then Exit Function
    PaybackProblem = modShiftSwitch.PaybackProblem(mOff.Text, mWorks.Text, d, d2, sh2, mShift.Text)     ' v25 review: the shift given up here
End Function

' first line of an EligibilityProblems list, without its "- "
Private Function FirstProblem(ByVal s As String) As String
    s = Trim$(s)
    If s = "" Then Exit Function
    Dim p As Long: p = InStr(s, vbNewLine)
    If p > 0 Then s = Left$(s, p - 1)
    If Left$(s, 2) = "- " Then s = Mid$(s, 3)
    FirstProblem = s
End Function

Private Sub mTwoWay_Click()
    mPayDate.Enabled = mTwoWay.Value
    RefillWorks
    If mTwoWay.Value Then
        C("cboPay").ListIndex = 0                ' a two-way swap is a new pair
        C("cboPay").Enabled = False
        mPayDate.SetFocus
    Else
        C("cboPay").Enabled = True
        mPayDate.Text = ""
        LoadPayback                          ' bring back any open payback for this pair
    End If
    Refresh False
End Sub

Private Sub mPayDate_Change()
    RefillWorks                          ' a same-day sister swap lists the other shift's crew
    Refresh False
End Sub

Private Sub SetHint(ByVal nm As String, ByVal txt As String, ByVal ok As Boolean)
    With C(nm)
        .Caption = txt
        .ForeColor = IIf(ok, RGB(0, 110, 60), RGB(190, 30, 30))
    End With
End Sub

Private Function ShortName(ByVal nm As String) As String
    ShortName = SwitchShortName(nm)                ' v25: the module's
End Function

' Payback choices: earlier switches where today's "works" gave up a shift
' and today's "off" worked it, not yet paid back.
Private Sub LoadPayback()
    Dim cb As Object: Set cb = C("cboPay")
    cb.Clear
    cb.AddItem "(none - this is a new switch)"
    mPayIds = Empty
    If mTwoWay.Value Then cb.ListIndex = 0: Exit Sub   ' a two-way swap is always a new pair
    If mOff.ListIndex >= 0 And mWorks.ListIndex >= 0 Then
        Dim d As Variant, i As Long
        d = OpenPaybacks(mOff.Text, mWorks.Text, mPayIds)
        If IsArray(d) Then
            For i = LBound(d) To UBound(d)
                cb.AddItem d(i)
            Next i
            cb.ListIndex = 1                 ' most likely: it's the payback
            If mWantPay <> "" Then
                For i = LBound(mPayIds) To UBound(mPayIds)
                    If StrComp(CStr(mPayIds(i)), mWantPay, vbTextCompare) = 0 Then cb.ListIndex = i + 1
                Next i
            End If
            Exit Sub
        End If
    End If
    cb.ListIndex = 0
End Sub

Private Sub LoadList()
    Dim v As Variant
    mList.Clear
    mListIds = Empty
    If mShowAll.Value Then
        C("hdr2").Caption = "RECENT SWITCHES (all)"
        v = RecentSwitches(60, mListIds)
        If IsArray(v) Then
            mList.List = v
            C("status").Caption = (UBound(v, 1) + 1) & " shown, newest first.  owed / paid SS-# / pays SS-# / VOID"
        Else
            C("status").Caption = "No switches logged yet."
        End If
    Else
        C("hdr2").Caption = "OWED - click one to log its payback"
        v = OwedSwitches(mListIds)
        If IsArray(v) Then
            mList.List = v
            C("status").Caption = (UBound(v, 1) + 1) & " switch(es) still owed a payback."
        Else
            C("status").Caption = "Nothing owed - every switch has been paid back."
        End If
    End If
End Sub

Private Sub mShowAll_Click()
    LoadList
End Sub


' ================================================================
' CLICK AN OWED SWITCH -> FILL THE PAYBACK
' ================================================================
Private Sub mList_Click()
    If mShowAll.Value Then Exit Sub                 ' history view: select only (for Void)
    Dim i As Long: i = mList.ListIndex
    If i < 0 Or Not IsArray(mListIds) Then Exit Sub
    Dim id As String: id = CStr(mListIds(i))
    Dim d0 As Date, sh As String, unit As String, offN As String, worksN As String
    If Not SwitchInfo(id, d0, sh, unit, offN, worksN) Then Exit Sub

    ' payback = the reverse: the one who worked now gives up a shift
    If mTwoWay.Value Then mTwoWay.Value = False
    mFilling = True
    On Error Resume Next
    mUnit.Value = unit
    mOff.ListIndex = -1: mWorks.ListIndex = -1
    FillPeople mOff, Array(worksN)           ' payback pair: keep them in the lists until the date is entered
    mOff.Value = worksN
    FillPeople mWorks, Array(offN)
    mWorks.Value = offN
    On Error GoTo 0
    mFilling = False
    mWantPay = id
    LoadPayback
    mWantPay = ""

    mDate.Text = ""
    C("txtNote").Text = ""
    C("status").Caption = ShortName(worksN) & " pays back " & ShortName(offN) & " (" & id & ")" & _
                          ": enter the payback date and save."
    mDate.SetFocus
End Sub

Private Sub mClear_Click()
    mFilling = True
    mOff.ListIndex = -1: mWorks.ListIndex = -1
    mShift.ListIndex = -1
    mFilling = False
    mDate.Text = "": C("txtNote").Text = ""
    mTwoWay.Value = False
    LoadPayback
    mList.ListIndex = -1
    C("status").Caption = ""
End Sub


' ================================================================
' SAVE / VOID
' ================================================================
Private Sub mSave_Click()
    Dim d As Date, problems As String
    If Not PickedDate(d) Then problems = problems & "- Shift date" & vbNewLine
    If C("cboUnit").ListIndex < 0 Then problems = problems & "- Unit" & vbNewLine
    If mOff.ListIndex < 0 Then problems = problems & "- Gives up shift" & vbNewLine
    If mWorks.ListIndex < 0 Then problems = problems & "- Works the shift" & vbNewLine
    If mShift.ListIndex < 0 Then problems = problems & "- Shift (Days / Nights)" & vbNewLine
    If problems <> "" Then MsgBox "Please fill in:" & vbNewLine & vbNewLine & problems, vbExclamation, "Shift Switch": Exit Sub
    If mOff.Text = mWorks.Text Then MsgBox "The same person is on both sides.", vbExclamation, "Call Out Roster": Exit Sub

    Dim isNight As Boolean: isNight = (mShift.Text = "Nights")
    Dim twoWay As Boolean: twoWay = mTwoWay.Value
    Dim d2 As Date, sh2 As String
    If twoWay Then
        If Not PayDateOk(d2) Then MsgBox "Enter the payback date (or untick Two-way swap).", vbExclamation, "Call Out Roster": Exit Sub
        sh2 = PayShift(d2)
        Dim payProb As String: payProb = PaybackProblem(d2, sh2)
        If payProb <> "" Then
            MsgBox "The payback on " & Format$(d2, "ddd mmm d") & " can't be logged:" & vbNewLine & vbNewLine & _
                   "- " & payProb & vbNewLine & vbNewLine & "Pick another payback date.", vbExclamation, "Not eligible"
            Exit Sub
        End If
        If Int(d2) = Int(d) And sh2 = mShift.Text Then
            MsgBox "The payback is the same date AND shift - that isn't a swap.", vbExclamation, "Call Out Roster": Exit Sub
        End If
    End If

    ' ---- hard stops: booked off / already on a switch (v25: modShiftSwitch.SwitchSaveStops) ----
    Dim stops As String
    stops = SwitchSaveStops(mOff.Text, mWorks.Text, d, mShift.Text, twoWay, d2, sh2)
    If stops <> "" Then
        MsgBox "This switch can't be logged:" & vbNewLine & vbNewLine & stops, vbExclamation, "Not eligible"
        Exit Sub
    End If

    ' ---- rotation warnings (v25: modShiftSwitch.SwitchWarnings) ----
    Dim warn As String
    warn = SwitchWarnings(mOff.Text, mWorks.Text, d, isNight, (twoWay And Int(d2) = Int(d)))

    Dim payFor As String
    If Not twoWay Then
        If C("cboPay").ListIndex > 0 And IsArray(mPayIds) Then payFor = CStr(mPayIds(C("cboPay").ListIndex - 1))
    End If

    Dim msg As String
    msg = Format$(d, "dddd mmm d, yyyy") & "  -  " & mShift.Text & "  -  " & C("cboUnit").Text & vbNewLine & vbNewLine & _
          mWorks.Text & "  works for  " & mOff.Text & vbNewLine
    If twoWay Then
        msg = msg & vbNewLine & "Payback " & Format$(d2, "dddd mmm d, yyyy") & "  -  " & sh2 & vbNewLine & _
              mOff.Text & "  works for  " & mWorks.Text & vbNewLine & "(both logged now - nothing left owed)" & vbNewLine
    Else
        msg = msg & IIf(payFor <> "", "Payback for " & payFor & " (it will be cleared from the owed list)", _
                                      "New switch - stays on the owed list until paid back") & vbNewLine
    End If
    If warn <> "" Then msg = msg & vbNewLine & "Rotation check:" & vbNewLine & warn
    If MsgBox(msg & vbNewLine & "Save?", vbOKCancel + IIf(warn <> "", vbExclamation, vbQuestion), "Confirm shift switch") <> vbOK Then Exit Sub

    Dim id As String, id2 As String
    id = SaveSwitch(d, mShift.Text, C("cboUnit").Text, mOff.Text, mWorks.Text, payFor, Trim$(C("txtNote").Text))
    If id = "" Then Exit Sub
    If twoWay Then
        id2 = SaveSwitch(d2, sh2, C("cboUnit").Text, mWorks.Text, mOff.Text, id, Trim$(C("txtNote").Text))
        If id2 = "" Then MsgBox "The switch was saved (" & id & ") but its payback wasn't - it stays on the owed list.", vbExclamation, "Call Out Roster"
        payFor = ""
    End If

    ' ready for the next one (keep date + unit)
    mFilling = True
    mOff.ListIndex = -1: mWorks.ListIndex = -1
    mFilling = False
    C("txtNote").Text = ""
    RefillPeople                         ' the two people just switched - rebuild who's on / free
    LoadPayback
    LoadList
    If twoWay And id2 <> "" Then
        C("status").Caption = "Saved: the switch and its payback (" & id & ", " & id2 & ")."
        mTwoWay.Value = False
    Else
        C("status").Caption = "Saved (" & id & ")" & IIf(payFor <> "", " - " & payFor & " is paid back.", " - owed until paid back.")
    End If
End Sub

Private Sub mVoid_Click()
    Dim i As Long: i = mList.ListIndex
    If i < 0 Or Not IsArray(mListIds) Then MsgBox "Select a switch in the list first.", vbInformation, "Call Out Roster": Exit Sub
    Dim id As String: id = CStr(mListIds(i))
    If VoidSwitch(id) Then
        ' in the owed view, clicking the row filled the form with its payback - clear it
        If mShowAll.Value Then RefillPeople Else mClear_Click
        LoadPayback
        LoadList
        C("status").Caption = "Voided (" & id & ")."
    End If
End Sub

Private Sub mClose_Click()
    Unload Me
End Sub
