' @version v27 (Oct 22)

' ================================================================
' SICK CALL (v22)
'  Books the sick shift (Absences, type SIC, id SC-####), which keeps the
'  person off call lists for that shift and any shift starting within
'  24 h after it ends, puts an OPEN line on the calendar, and opens Log
'  Call-Out already filled in so the supervisor can start calling.
'  Every control is built in code below (new form: Insert > UserForm,
'  name it frmSickCall, paste this).
' ================================================================

Private WithEvents mEmp As MSForms.ComboBox
Private WithEvents mDate As MSForms.TextBox
Private WithEvents mShift As MSForms.ComboBox
Private WithEvents mSave As MSForms.CommandButton
Private WithEvents mSaveOnly As MSForms.CommandButton
Private WithEvents mClose As MSForms.CommandButton

Private Const LBL_L As Long = 14
Private Const BOX_L As Long = 118
Private Const ROW_H As Long = 26

Private Function C(ByVal nm As String) As Object
    Set C = Me.Controls("sc_" & nm)
End Function

Private Function AddCtl(ByVal progId As String, ByVal nm As String, ByVal L As Single, ByVal T As Single, _
                        ByVal W As Single, ByVal H As Single) As Object
    Dim o As Object
    Set o = Me.Controls.Add(progId, "sc_" & nm, True)
    o.Left = L: o.Top = T: o.Width = W: o.Height = H
    Set AddCtl = o
End Function

Private Function AddLabel(ByVal nm As String, ByVal cap As String, ByVal L As Single, ByVal T As Single, ByVal W As Single) As Object
    Dim o As Object
    Set o = AddCtl("Forms.Label.1", nm, L, T, W, 15)
    o.Caption = cap
    Set AddLabel = o
End Function

Private Function RowT(ByVal i As Long) As Single
    RowT = 36 + (i - 1) * ROW_H
End Function

Private Sub FillCombo(ByVal cb As Object, ByVal items As Variant)
    Dim i As Long
    cb.Clear
    If Not IsArray(items) Then Exit Sub
    For i = LBound(items) To UBound(items)
        cb.AddItem CStr(items(i))
    Next i
End Sub

' ---- live defaults from the rotation and the roster ----
Private Sub mEmp_Change()
    Refresh
End Sub

Private Sub mDate_Change()
    Refresh
End Sub

Private Sub mShift_Change()
    Dim d As Date
    If PickedDate(d) Then
        C("txtHours").Text = CStr(DefaultShiftHours(d, IsNightShift(mShift.Text), USUAL_SHIFT_HOURS))
        C("hHrs").Caption = IIf(IsNightShift(mShift.Text), DstNote(d), "")
        ' v26.5.5 (B1): the night is named by the evening it starts; before 7 AM the shift under way is said
        Dim hint As String: hint = ShiftHintText(d, IsNightShift(mShift.Text), Now)
        C("info").Caption = "Off call lists for this shift and any shift starting within 24 h after it ends." & _
                            IIf(hint <> "", vbLf & hint & ".", "")
        ShowCounters d
    End If
End Sub

' v22.1: the person's sick / personal counters and what this shift would be
Private Sub ShowCounters(ByVal d As Date)
    If mEmp.ListIndex < 0 Or mShift.ListIndex < 0 Then C("hAtt").Caption = "": Exit Sub
    On Error Resume Next
    C("hAtt").Caption = SickCallPreview(mEmp.Text, d, mShift.Text)
    If Err.Number <> 0 Then C("hAtt").Caption = "Counters not available: " & Err.Description
    On Error GoTo 0
End Sub

Private Function PickedDate(ByRef d As Date) As Boolean
    PickedDate = ParseTypedDate(mDate.Text, d)      ' v25: the one parser
End Function

Private Sub Refresh()
    Dim d As Date
    If Not PickedDate(d) Then
        C("hDate").Caption = IIf(Trim$(mDate.Text) = "", "", "not a date")
        Exit Sub
    End If
    C("hDate").Caption = "= " & Format$(d, "ddd mmm d, yyyy")
    If mEmp.ListIndex < 0 Then mShift_Change: Exit Sub
    Dim sch As String: sch = ScheduledShift(mEmp.Text, d)
    If sch <> "" Then
        mShift.Value = sch
        C("hShift").Caption = "scheduled " & LCase$(sch) & " (" & CrewStatus(mEmp.Text, d) & ")"
    Else
        If mShift.ListIndex < 0 Then mShift.Value = "Days"
        C("hShift").Caption = "not scheduled that day - pick the shift they were to work"
    End If
    Dim hu As String: hu = HomeUnitOf(mEmp.Text)
    On Error Resume Next                         ' a Home Unit not in the list leaves the box as it is
    If hu <> "" Then C("cboUnit").Value = hu
    On Error GoTo 0
    mShift_Change
End Sub

' ---- save ----
Private Function SaveIt() As String
    Dim d As Date, problems As String
    If mEmp.ListIndex < 0 Then problems = problems & "- Who called in" & vbNewLine
    If Not PickedDate(d) Then problems = problems & "- Shift date" & vbNewLine
    If mShift.ListIndex < 0 Then problems = problems & "- Shift" & vbNewLine
    If C("cboUnit").ListIndex < 0 Then problems = problems & "- Unit" & vbNewLine
    Dim h As String: h = Trim$(C("txtHours").Text)
    If h = "" Or Not IsNumeric(h) Then problems = problems & "- Hours" & vbNewLine
    If problems <> "" Then MsgBox "Please fill in:" & vbNewLine & vbNewLine & problems, vbExclamation, "Sick Call": Exit Function
    If HasSickRow(mEmp.Text, d, mShift.Text) Then            ' v22.1: already logged (e.g. Save only, then reopened)
        MsgBox mEmp.Text & " is already booked sick for " & Format$(d, "ddd mmm d") & " " & mShift.Text & "." & _
               IIf(IsAdmin(), " Void that booking in Book Time Off if it was wrong.", " If it was wrong, ask " & PRIMARY_ADMIN_NAME & " to void it."), _
               vbExclamation, "Sick Call"       ' v26.5.5 (B10): voiding is an admin's job
        Exit Function
    End If
    If Abs(Int(d) - Int(Date)) > 7 Then
        If MsgBox("That date is " & Abs(Int(d) - Int(Date)) & " days from today. Save it anyway?", vbYesNo + vbQuestion, "Call Out Roster") = vbNo Then Exit Function
    End If
    Dim hrs As Variant: hrs = CDbl(h)
    ' a full shift (12 h, or the 11 h / 13 h DST night) gets no partial-hours mark on the line
    If CDbl(h) >= DefaultShiftHours(d, IsNightShift(mShift.Text), USUAL_SHIFT_HOURS) Then hrs = Empty
    ' v22.1: the note records what the rule made of this shift (paid / unpaid and why)
    Dim note As String, outcome As String
    note = Trim$(C("txtNote").Text)
    On Error Resume Next
    outcome = SickOutcome(mEmp.Text, d, mShift.Text)
    On Error GoTo 0
    If outcome <> "" Then note = note & IIf(note = "", "", " - ") & outcome
    SaveIt = SaveSickCall(mEmp.Text, d, mShift.Text, C("cboUnit").Text, hrs, note)
End Function

Private Sub mSaveOnly_Click()
    Dim id As String: id = SaveIt()
    If id = "" Then Exit Sub
    Dim d0 As Date: PickedDate d0
    ' v27 (item 7): the receipt card (v26.5.5's box said the same)
    ShowReceiptNow "Sick call " & id & " saved", _
                   "  " & id & "  Sick  " & mEmp.Text & "  " & Format$(d0, "ddd mmm d") & " " & mShift.Text & "  (an Absences row; no OT Log row)" & vbNewLine, _
                   "  Nobody's Total Hours changed. " & mEmp.Text & " is off call lists for this shift and the 24 h after it." & vbNewLine, _
                   "  Calendar: " & Format$(d0, "ddd mmm d") & " " & mShift.Text & ", " & C("cboUnit").Text & " - the line is OPEN (Fill a shift now, or Post a shift)." & vbNewLine & _
                   "  Attendance (admin): the sick / personal counters." & vbNewLine & "  The members' file shows " & mEmp.Text & " as Off after the next update." & vbNewLine, _
                   IIf(IsAdmin(), "  Book Time Off > Void this booking (or the calendar line's menu) voids it." & vbNewLine, "  Wrong person or shift? Ask " & PRIMARY_ADMIN_NAME & " to void it." & vbNewLine)
    Unload Me
End Sub

Private Sub mSave_Click()
    Dim id As String: id = SaveIt()
    If id = "" Then Exit Sub
    Dim unit As String: unit = C("cboUnit").Text
    Dim d As Date: If Not PickedDate(d) Then Exit Sub
    Dim sh As String: sh = mShift.Text
    Dim hv As Double: hv = Val(C("txtHours").Text)           ' v26.5.5 (B8): the sick call's hours go to the call-out
    Unload Me
    If Not RequireWritable() Then Exit Sub
    If LogTable() Is Nothing Then Exit Sub
    Load frmOvertimeEntry
    frmOvertimeEntry.Prefill unit, d, sh, "", hv, "A:" & id     ' v27 (item 10): the acceptance covers this sick call's line, no question
    frmOvertimeEntry.Show
End Sub

Private Sub mClose_Click()
    Unload Me
End Sub
