' ================================================================
' BOOK TIME OFF  /  ADD A NEED
'  Person off : vacation, personal day, training, leave (LOA / STD /
'               parental / WSIB / LTD) or a past sick day, for a date
'               range. End can be left open (STD with no return date)
'               and set later. The rotation decides which days / nights
'               it shows on. Leave types show as "Leave".
'               v22.1: vacation and personal days are booked on scheduled
'               shifts only (no shifts in the range = refused; days off
'               at either end = offer to trim). "SIC - Sick day (past)"
'               records one sick shift per scheduled shift in the range.
'  Need       : a line to fill with nobody off (PHA, support op ...)
'               for each date in a range, Days / Nights / Both.
'
' Every control is built in code below. Anything drawn on the form in
' the designer is hidden automatically.
' ================================================================

Private Const LBL_L As Long = 14

' ================================================================
' CONTROL HELPERS (runtime names start "to_")
' ================================================================
Private Function C(ByVal nm As String) As Object
    Set C = Me.Controls("to_" & nm)
End Function

Private Function AddCtl(ByVal progId As String, ByVal nm As String, ByVal L As Single, ByVal T As Single, _
                        ByVal W As Single, ByVal H As Single, Optional ByVal grp As String = "") As Object
    Dim o As Object
    Set o = Me.Controls.Add(progId, "to_" & nm, True)
    o.Left = L: o.Top = T: o.Width = W: o.Height = H
    If grp = "P" Then mPersonCtls.Add o
    If grp = "N" Then mNeedCtls.Add o
    Set AddCtl = o
End Function

Private Function AddLabel(ByVal nm As String, ByVal cap As String, ByVal L As Single, ByVal T As Single, _
                          ByVal W As Single, Optional ByVal grp As String = "") As Object
    Dim o As Object
    Set o = AddCtl("Forms.Label.1", nm, L, T, W, 15, grp)
    o.Caption = cap
    Set AddLabel = o
End Function

Private Function RowT(ByVal i As Long) As Single
    RowT = 40 + (i - 1) * ROW_H
End Function

Private Sub FillCombo(ByVal cb As Object, ByVal items As Variant)
    Dim i As Long
    cb.Clear
    If Not IsArray(items) Then Exit Sub
    For i = LBound(items) To UBound(items)
        cb.AddItem CStr(items(i))
    Next i
End Sub

Private Sub SetMode(ByVal person As Boolean)
    Dim o As Object
    For Each o In mPersonCtls
        o.Visible = person
    Next o
    For Each o In mNeedCtls
        o.Visible = Not person
    Next o
    Me.Caption = IIf(person, "Book Time Off", "Add a Need")
    Me.Height = UiPx(IIf(person, 560, RowT(11) + 74))     ' v25.5: after ScaleForm, so x UI_SCALE here
    UpdateInfo
End Sub

Private Sub mEmp_Change()
    LoadList

    DateHint mEnd, "hEnd"

Private Sub mOpen_Click()
    mEnd.Enabled = Not mOpen.Value
    If mOpen.Value Then mEnd.Text = ""
    UpdateInfo
End Sub

Private Sub DateHint(ByVal tb As Object, ByVal nm As String)
    TypedDateHint tb.Text, C(nm)                   ' v25: the one parser's hint
End Sub

Private Sub UpdateInfo()
    Dim lbl As Object: Set lbl = C("info")
    lbl.ForeColor = RGB(30, 56, 100)
    Dim d1 As Date, d2 As Date
    If Not ParseTypedDate(mStart.Text, d1) Then lbl.Caption = "": Exit Sub
    If mOptNeed.Value Then
        If Not ParseTypedDate(mEnd.Text, d2) Then d2 = d1
        lbl.Caption = (Int(d2) - Int(d1) + 1) & " day(s) - one line per day" & _
                      IIf(C("cboShift").Text = "Both", " and shift", "") & "."
        Exit Sub
    End If
    If mEmp.ListIndex < 0 Then lbl.Caption = "": Exit Sub
    Dim nD As Long, nN As Long
    If mOpen.Value Then
        ShiftCount mEmp.Text, d1, d1 + 27, nD, nN
        lbl.Caption = "Open-ended from " & Format$(d1, "mmm d") & ": " & nD & " day / " & nN & _
                      " night shifts in the first 4 weeks. Set the end date when known."
    ElseIf ParseTypedDate(mEnd.Text, d2) Then
        If d2 < d1 Then lbl.Caption = "The end is before the start.": lbl.ForeColor = RGB(190, 30, 30): Exit Sub
        ShiftCount mEmp.Text, d1, d2, nD, nN
        lbl.Caption = "Covers " & (nD + nN) & " scheduled shift(s): " & nD & " day, " & nN & " night." & _
                      IIf(nD + nN = 0, "  (none - check the dates / crew)", "")
    Else
        lbl.Caption = ""
    End If
End Sub

Private Sub LoadList()
    C("lst").Clear
    mIds = Empty
    If mEmp.ListIndex < 0 Then C("status").Caption = "": Exit Sub
    Dim v As Variant: v = PersonAbsences(mEmp.Text, mIds)
    If IsArray(v) Then
        C("lst").List = v
        C("status").Caption = (UBound(v, 1) + 1) & " booking(s) for " & mEmp.Text & ", newest first."
    Else
        C("status").Caption = "Nothing booked for " & mEmp.Text & "."
    End If
End Sub

Private Function HoursValue() As Variant
    Dim t As String: t = Trim$(C("cboHrs").Text)
    HoursValue = Empty
    If t = "" Then Exit Function
    If Val(t) > 0 And Val(t) < 12 Then HoursValue = Val(t)
End Function

Private Sub mSave_Click()
    Dim problems As String
    If C("cboUnit").ListIndex < 0 Then problems = problems & "- Unit" & vbNewLine
    Dim ds0 As Date, de0 As Date                   ' v25: every date through the one parser
    If Not ParseTypedDate(mStart.Text, ds0) Then problems = problems & "- From date" & vbNewLine
    Dim t As String: t = Trim$(C("cboHrs").Text)
    If t <> "" And Val(t) <= 0 Then problems = problems & "- Hours / shift" & vbNewLine

    If mOptNeed.Value Then
        If Trim$(C("cboNeed").Text) = "" Then problems = problems & "- Need" & vbNewLine
        If Trim$(mEnd.Text) <> "" And Not ParseTypedDate(mEnd.Text, de0) Then problems = problems & "- To date" & vbNewLine
        If problems <> "" Then MsgBox "Please fill in:" & vbNewLine & vbNewLine & problems, vbExclamation, "Call Out Roster": Exit Sub
        Dim d1 As Date, d2 As Date: d1 = ds0
        If Trim$(mEnd.Text) <> "" Then d2 = de0 Else d2 = d1
        If d2 < d1 Then MsgBox "The end is before the start.", vbExclamation, "Call Out Roster": Exit Sub
        Dim needTxt As String, typ As String
        If C("cboNeed").ListIndex >= 0 Then
            needTxt = Trim$(Mid$(C("cboNeed").Text, 7))
            typ = Left$(C("cboNeed").Text, 3)
        Else
            needTxt = Trim$(C("cboNeed").Text)
            If InStr(1, needTxt, "PHA", vbTextCompare) > 0 Then
                typ = "PHA"
            ElseIf InStr(1, needTxt, "train", vbTextCompare) > 0 Then
                typ = "TRN"
            Else
                typ = "SUP"
            End If
        End If
        If Trim$(C("txtNote").Text) <> "" Then needTxt = needTxt & " - " & Trim$(C("txtNote").Text)
        Dim nLines As Long
        nLines = SaveNeed(C("cboUnit").Text, needTxt, typ, d1, d2, C("cboShift").Text, HoursValue())
        If nLines > 0 Then C("status").Caption = nLines & " need line(s) added."
        If nLines > 0 Then MsgBox nLines & " line(s) added to the " & C("cboUnit").Text & " calendar.", vbInformation, "Call Out Roster"
        Exit Sub
    End If

    ' ---- person off ----
    If mEmp.ListIndex < 0 Then problems = problems & "- Employee" & vbNewLine
    If C("cboType").ListIndex < 0 Then problems = problems & "- Type" & vbNewLine
    If Not mOpen.Value And Not ParseTypedDate(mEnd.Text, de0) Then problems = problems & "- To date (or tick Open-ended)" & vbNewLine
    If problems <> "" Then MsgBox "Please fill in:" & vbNewLine & vbNewLine & problems, vbExclamation, "Call Out Roster": Exit Sub

    Dim ds As Date: ds = ds0
    Dim de As Variant
    If mOpen.Value Then
        de = Empty
    Else
        de = de0
        If de < ds Then MsgBox "The end is before the start.", vbExclamation, "Call Out Roster": Exit Sub
    End If
    If ds < RosterYearFrom() Then                    ' v25: a year typed wrong (2008 for 2026) is caught here
        MsgBox "The start (" & Format$(ds, "mmm d, yyyy") & ") is before this roster year's first day (" & _
               Format$(RosterYearFrom(), "mmm d, yyyy") & "). Check the year.", vbExclamation, "Book Time Off"
        Exit Sub
    End If

    Dim code As String: code = TypeCode(C("cboType").Text)
    Dim id As String
    If code = "PER" And mOpen.Value Then MsgBox "Personal days need an end date (each scheduled shift counts as one day).", vbExclamation, "Call Out Roster": Exit Sub
    If code = "SIC" Then                          ' v22.1: past sick days, one row per scheduled shift
        If mOpen.Value Then MsgBox "Past sick days need an end date.", vbExclamation, "Call Out Roster": Exit Sub
        If ds > Date Then MsgBox "Past sick days are for days already gone (" & Format$(ds, "mmm d, yyyy") & " is ahead). For today or later use Sick Call.", vbExclamation, "Call Out Roster": Exit Sub
        Dim nSick As Long, nUnsched As Long, nAlready As Long, skipped As String
        nSick = SaveSickPast(mEmp.Text, C("cboUnit").Text, ds, CDate(de), Trim$(C("txtNote").Text), HoursValue(), nUnsched, nAlready)
        If nUnsched > 0 Then skipped = nUnsched & " day(s) with no scheduled shift"
        If nAlready > 0 Then skipped = skipped & IIf(skipped = "", "", ", ") & nAlready & " already booked sick"
        If nSick = 0 Then                               ' v25 (L6): says which it was
            MsgBox "No sick days were recorded for " & mEmp.Text & ", " & Format$(ds, "mmm d") & " to " & Format$(de, "mmm d") & ": " & _
                   IIf(skipped = "", "the file is read-only, or the dates are before this roster year", skipped) & ".", vbExclamation, "Call Out Roster"
            Exit Sub
        End If
        LoadList
        C("status").Caption = nSick & " past sick day(s) recorded for " & mEmp.Text & ", " & _
                              Format$(ds, "mmm d") & " to " & Format$(de, "mmm d") & IIf(skipped = "", ".", " (skipped: " & skipped & ").")
        mStart.Text = "": mEnd.Text = "": C("txtTimes").Text = "": C("txtNote").Text = ""
        Exit Sub
    End If
    If (code = "VAC" Or code = "PER") And Not mOpen.Value Then      ' v22.1: scheduled shifts only (v25: modCalendar.ScheduledSpan / TrimKind)
        Dim nD As Long, nN As Long, dFirst As Date, dLast As Date
        If ScheduledSpan(mEmp.Text, ds, CDate(de), dFirst, dLast, nD, nN) = 0 Then
            MsgBox mEmp.Text & " has no scheduled shift between " & Format$(ds, "mmm d") & " and " & Format$(de, "mmm d") & "." & vbNewLine & vbNewLine & _
                   "Vacation and personal days are booked on scheduled shifts only - check the dates or their crew.", vbExclamation, "Days off"
            Exit Sub
        End If
        If TrimKind(ds, CDate(de), dFirst, dLast) <> "" Then
            Select Case MsgBox(Format$(ds, "mmm d") & " to " & Format$(de, "mmm d") & " covers " & (nD + nN) & " scheduled shift(s); " & _
                               IIf(dFirst > ds, Format$(ds, "mmm d") & IIf(dFirst - ds > 1, " - " & Format$(dFirst - 1, "mmm d"), "") & " at the start", "") & _
                               IIf(dFirst > ds And dLast < CDate(de), " and ", "") & _
                               IIf(dLast < CDate(de), Format$(dLast + 1, "mmm d") & IIf(CDate(de) - dLast > 1, " - " & Format$(de, "mmm d"), "") & " at the end", "") & _
                               " are days off." & vbNewLine & vbNewLine & _
                               "Trim the booking to " & Format$(dFirst, "mmm d") & " - " & Format$(dLast, "mmm d") & "?" & vbNewLine & _
                               "Yes = trim it, No = save as typed, Cancel = go back", vbYesNoCancel + vbQuestion, "Days off in the range")
                Case vbYes: ds = dFirst: de = dLast
                Case vbCancel: Exit Sub
            End Select
        End If
    End If
    id = SaveAbsence(mEmp.Text, C("cboUnit").Text, C("cboType").Text, ds, de, HoursValue(), _
                     Trim$(C("txtTimes").Text), Trim$(C("txtNote").Text))
    If id = "" Then Exit Sub
    LoadList
    C("status").Caption = "Saved: " & mEmp.Text & ", " & Left$(C("cboType").Text, 3) & ", " & _
                          Format$(ds, "mmm d") & " to " & IIf(mOpen.Value, "open", Format$(de, "mmm d")) & "  (" & id & ")."
    ' ready for the next booking
    mStart.Text = "": mEnd.Text = "": C("txtTimes").Text = "": C("txtNote").Text = ""
    mOpen.Value = False
End Sub

Private Function PickedId() As String
    Dim i As Long: i = C("lst").ListIndex
    If i < 0 Or Not IsArray(mIds) Then MsgBox "Select a booking in the list first.", vbInformation, "Call Out Roster": Exit Function
    PickedId = CStr(mIds(i))
End Function

Private Sub mSetEnd_Click()
    Dim id As String: id = PickedId()
    If id = "" Then Exit Sub
    If SetAbsenceEnd(id) Then LoadList: C("status").Caption = "End date changed (" & id & ")."
End Sub

Private Sub mVoid_Click()
    Dim id As String: id = PickedId()
    If id = "" Then Exit Sub
    If VoidAbsence(id) Then LoadList: C("status").Caption = "Voided (" & id & ")."
End Sub
