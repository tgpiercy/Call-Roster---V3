' FIX AN ENTRY (v29, item 4; briefs/admin-lite.md): the guided correction
'  for the Admin Lite Desk's Corrections card, for any admin. Pick a Ref ID
'  from the receipts answered No / Not sure, or type one; see every row it
'  wrote and whose Total Hours it moved; Void with a reason (the existing

'  built in code below (new form: Insert > UserForm, name it frmFixEntry,
'  paste this).
' ================================================================

Private Function PickedRef() As String
    Dim s As String: s = Trim$(mRef.Text)
    If s = "" And mPick.ListIndex >= 0 And mPick.Enabled Then
        s = mPick.List(mPick.ListIndex)
        If InStr(s, "  -  ") > 0 Then s = Left$(s, InStr(s, "  -  ") - 1)
    End If
    PickedRef = UCase$(Trim$(s))
End Function

Private Sub ShowFacts()
    Dim ref As String: ref = PickedRef()
    If ref = "" Then mView.Text = "Pick an entry above, or type its Ref ID.": Exit Sub
    If mVoided And ref = mFacts.Ref Then Exit Sub            ' (review MED-2) the voided one is still on screen: nothing to reload
    If ref <> mFacts.Ref Then ResolveIfLeft                  ' moving on from the one shown: its answer is marked (voided / checked)
    mFacts = FixEntryFacts(ref)
    mVoided = False: mResolved = False
    mView.Text = Replace(mFacts.Text, vbNewLine, vbCrLf)
    mVoid.Enabled = mFacts.Found And mFacts.Kind <> "P"
    mReenter.Enabled = False
    If mFacts.Kind = "P" And mFacts.Found Then
        mHint.Caption = "An award: use Withdraw an award / edit a posting on the full Desk (the refusals stay charged; a call-out is offered after)."
    ElseIf mFacts.Found Then
        mHint.Caption = "Step 2: type the reason, then Void this entry. Close instead marks the receipt's answer as checked (the entry stands)."
    Else
        mHint.Caption = "Nothing live to void (a typo, or already voided). Close marks the receipt's answer as looked at."
    End If
End Sub

Private Sub mPick_Change()
    mRef.Text = ""
    ShowFacts
End Sub

Private Sub mShow_Click()
    ShowFacts
End Sub

Private Sub mVoid_Click()
    On Error GoTo Oops
    If Not mFacts.Found Then Exit Sub
    Dim why As String: why = Trim$(mWhy.Text)
    If why = "" Then mHint.Caption = "A reason is required before the void.": mWhy.SetFocus: Exit Sub
    If Ask("Void " & mFacts.Ref & "?" & vbNewLine & vbNewLine & Left$(mFacts.Text, 400) & vbNewLine & vbNewLine & _
           "Reason: " & why & vbNewLine & "This cannot be undone (a new entry replaces it).", vbYesNo + vbQuestion, "Fix an entry", vbYes) <> vbYes Then Exit Sub
    Dim msg As String
    Me.Hide                                          ' the void's own boxes show over the roster
    If FixEntryVoid(mFacts, why, msg) Then
        mVoided = True
        mView.Text = "VOIDED: " & msg & vbCrLf & vbCrLf & Replace(mFacts.Text, vbNewLine, vbCrLf)
        mVoid.Enabled = False
        mReenter.Enabled = True
        mHint.Caption = "Step 3: Re-enter it now? opens the right pop-up pre-filled. Close instead marks the receipt's answer resolved as voided."
    Else
        mHint.Caption = msg
    End If
    Me.Show
    Exit Sub
Oops:
    ReportError "frmFixEntry.Void"                   ' (review MED-3) first in the handler; it closes the pop-ups itself
End Sub

Private Sub mReenter_Click()
    On Error GoTo Oops
    If Not mVoided Then Exit Sub
    Dim msg As String, newRef As String, emR As String
    Dim f As FixFacts: f = mFacts                    ' (review LOW-18) a copy: the re-entry pop-up's error handler may unload this form
    mResolved = True                                 ' whatever happens next, QueryClose must not resolve it a second time
    Me.Hide
    Dim rec0 As String: rec0 = gReceipt.Recorded     ' (re-review R1) the receipt before: an error AFTER the re-entry saved must not say "NOT re-entered"
    newRef = FixEntryReenter(f, msg)
    FixEntryDone f.Ref, newRef
    Tell IIf(newRef <> "", f.Ref & " was voided and re-entered as " & newRef & ". The receipt's answer is marked resolved with " & newRef & ".", _
             f.Ref & " is voided; nothing re-entered" & IIf(msg <> "", " - " & msg, ".") & " The receipt's answer is marked resolved as voided."), _
         vbInformation, "Fix an entry"
    Unload Me
    Exit Sub
Oops:
    emR = Err.Description
    On Error Resume Next                             ' (re-review LOW-4 / LOW-5) the pop-up raised: the request id must not ride on
    gRequestId = ""
    ReportError "frmFixEntry.Reenter"
    ' v29.0.3 (review 1.9): the feedback is NOT resolved on this path (the entry is void and nothing replaced it: the
    ' Corrections card keeps it until it is re-entered or looked at again), and the box names the voided Ref.
    ' (re-review R1) unless the re-entry DID save before the error (a receipt was recorded): then it is resolved with
    ' the new Ref and the box says so, or the person would enter it twice
    If f.Ref <> "" Then
        If gReceipt.Recorded <> rec0 And gReceipt.Recorded <> "" Then
            newRef = FirstRefIn(gReceipt.Recorded)
            If newRef <> "" Then FixEntryDone f.Ref, newRef
            MsgBox f.Ref & " was voided and RE-ENTERED as " & IIf(newRef <> "", newRef, "a new entry") & " - then something went wrong (" & emR & "). " & _
                   "Do NOT enter it again: check the new entry on the OT Log / Calendar.", vbExclamation, "Fix an entry"
        Else
            MsgBox f.Ref & " is VOID and was NOT re-entered - the re-entry stopped (" & emR & ")." & vbNewLine & vbNewLine & _
                   "Enter it again from its own button (Log Call-Out, Book Time Off, the Shift Switch pop-up ...). " & _
                   "Its receipt answer stays on the Corrections card until then.", vbExclamation, "Fix an entry"
        End If
    End If
End Sub

' the entry on screen when the card moves on or closes: a voided one nobody re-entered is resolved as "void"; one looked
' at and left as it stands as "checked" (or "already void" when nothing live was found) - review MED-1: the card's
' Corrections count must go down once the admin has looked. An award (P-) is left for Withdraw an award (re-review LOW-8).
Private Sub ResolveIfLeft()
    If mResolved Or mFacts.Ref = "" Then Exit Sub
    If mFacts.Kind = "P" And mFacts.Found And Not mVoided Then Exit Sub   ' a live award: Withdraw an award resolves it ("withdrawn"); a P- not found is "already void"
    FixEntryDone mFacts.Ref, IIf(mVoided, "", IIf(mFacts.Found, "checked", "already void"))
    mResolved = True
End Sub

Private Sub mClose_Click()
    ResolveIfLeft
    Unload Me
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If gErrorClosing Then Exit Sub                   ' (re-review MED-3) ReportError is unloading the pop-ups: nothing is resolved on an error path
    ResolveIfLeft
End Sub
