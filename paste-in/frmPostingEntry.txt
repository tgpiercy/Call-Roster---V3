' ================================================================
' AWARD OVERTIME POSTING
' Pick an open posting, tick who signed up (from the paper sheet),
' pick who is awarded, then one of:
'   SUBMIT AWARD   - anyone AHEAD of the (last) awarded person in call
'                    order who did NOT sign up is charged hours x 1.75.
'                    Signed up, below the awarded person, or awarded:
'                    not charged.
'   NO TAKERS      - nobody took it: everyone eligible is charged.
'   CANCEL REQUEST - overtime no longer needed: nobody is charged.
'   AWARD OUT OF DEPT - nobody in the department took it; award to a
'                    name from another department. In-department people
'                    who didn't sign up are charged as for NO TAKERS.
' PARTIAL HOURS (v24): someone signed up for only part of the shift.
'   Select the name, click PARTIAL HOURS, enter the hours. Full-shift
'   sign-ups are awarded first; a partial is awarded only when no
'   full-shift sign-up is left. Charging doesn't change: people ahead who
'   didn't sign up at all are charged the posting's full hours.
' Charges are previewed live. Uses modPostings for the rules and saving.
' (Design-time controls reused: btnLoad, lstEmployees, btnSignedUp,
'  btnPassed = AWARD toggle, btnSubmit. PARTIAL HOURS / NO TAKERS /
'  CANCEL REQUEST are added in code.)
' ================================================================

Private Sub LoadChoices()
    Dim d As Variant
    C("cmbPosting").Clear
    d = OpenPostingChoices(mIds)
    If IsArray(d) Then
        Dim i As Long
        For i = LBound(d) To UBound(d)
            C("cmbPosting").AddItem d(i)
        Next i
        C("lblStatus").Caption = (UBound(d) - LBound(d) + 1) & " open posting(s)."
    Else
        C("lblStatus").Caption = "No open postings."
    End If
End Sub


' ================================================================
' LOAD
' ================================================================
Private Sub btnLoad_Click()
    If mDirty Then
        If MsgBox("Discard the sign-ups / awards you've marked?", vbYesNo + vbExclamation) = vbNo Then Exit Sub
    End If
    Dim idx As Long: idx = C("cmbPosting").ListIndex
    If idx < 0 Or Not IsArray(mIds) Then MsgBox "Pick a posting first.", vbExclamation: Exit Sub

    mN = LoadPosting(CStr(mIds(idx)), mP, mL)
    If mN = 0 Then MsgBox "That posting has no list.", vbExclamation: Exit Sub
    mDirty = False

    C("lblInfo").Caption = mP.ID & "   " & mP.Unit & "   -   " & Format$(mP.ShiftDate, "dddd mmm d, yyyy") & vbLf & _
        "Shift: " & mP.ShiftLbl & "   Crew: " & mP.Crew & "   Hours: " & mP.Hours & _
        "   Positions: " & mP.Positions & vbLf & _
        "Reason: " & mP.Reason & "   |   Posted " & Format$(mP.PostedOn, "mmm d") & _
        ", closes " & Format$(mP.ClosesOn, "mmm d") & IIf(PostingClosed(mP.ClosesOn), "  (closed)", "  (STILL OPEN)")     ' v25 (P6): the one closing test

    SetActionButtons True
    FillOutside
    C("txtOutHrs").Text = ""
    ' v24.1: electronic sign-ups from the members' workbook are ticked for you to check
    Dim sTxt As String
    If ApplySignups(mP, mL, mN, sTxt) > 0 Then mDirty = True
    RefreshList 0
    If sTxt <> "" Then C("lblStatus").Caption = sTxt
End Sub

' signed off in this unit, not on the posting's list, and free that shift
Private Sub FillOutside()
    Dim cand As Variant, i As Long, cnt As Long
    cand = OutsideCandidates(mP, mL, mN)
    With C("cmbOutside")
        .Clear
        .Text = ""
        If IsArray(cand) Then
            For i = LBound(cand, 1) To UBound(cand, 1)
                .AddItem cand(i, 0)
                .List(.ListCount - 1, 1) = cand(i, 1)
                .List(.ListCount - 1, 2) = cand(i, 2)
                .List(.ListCount - 1, 3) = cand(i, 3)
            Next i
            cnt = UBound(cand, 1) - LBound(cand, 1) + 1
        End If
    End With
    C("lblOutside").Caption = "Out of department (signed a blank line):   " & _
        IIf(Not IsAdmin(), "an admin awards out of department (" & cnt & " free)", _
        IIf(cnt = 0, "nobody else signed off in " & mP.Unit & " is free that shift", _
                     cnt & " signed off in " & mP.Unit & " and free - lowest hours first"))
End Sub


' ================================================================
' LIST + LIVE PREVIEW
' ================================================================
Private Sub RefreshList(ByVal keepIdx As Long)
    Dim maxRank As Long, maxHrs As Double, nAw As Long, nCh As Long, tot As Double
    ComputeAward mP, mL, mN, maxRank, maxHrs, nAw, nCh, tot

    lstEmployees.Clear
    Dim i As Long, res As String
    For i = 1 To mN
        lstEmployees.AddItem CStr(mL(i).Rank)
        lstEmployees.List(i - 1, 1) = mL(i).EmpName
        lstEmployees.List(i - 1, 2) = mL(i).EmpShift
        lstEmployees.List(i - 1, 3) = Format$(mL(i).TotalHrs, "#,##0")
        If Not mL(i).SignedUp Then
            lstEmployees.List(i - 1, 4) = ""
        ElseIf IsPartialSignup(mP.Hours, mL(i).SignupHrs) Then
            lstEmployees.List(i - 1, 4) = HoursText(mL(i).SignupHrs) & " h"
        Else
            lstEmployees.List(i - 1, 4) = "Yes"
        End If
        Select Case mL(i).Result
            Case "Awarded"
                res = "AWARDED"
                If IsPartialSignup(mP.Hours, mL(i).SignupHrs) Then res = res & "  " & HoursText(mL(i).SignupHrs) & " h"
            Case "Charged": res = "REFUSED  " & HoursText(mL(i).Charged)
            Case "Signed up": res = "not charged"
            Case Else
                res = ""
                If mL(i).ExemptNote <> "" Then res = LCase$(Mid$(mL(i).ExemptNote, Len(EXEMPT_TAG) + 1))     ' "Exempt - moved Jan 12" -> "moved jan 12"
        End Select
        lstEmployees.List(i - 1, 5) = res
    Next i
    If keepIdx >= 0 And keepIdx < mN Then lstEmployees.ListIndex = keepIdx

    If nAw = 0 Then
        C("lblSummary").Caption = "No one awarded yet." & IIf(mP.Positions > 1, "  (" & mP.Positions & " positions)", "")
    Else
        Dim covered As Double, uncovered As Double, who As String, coverTxt As String
        AwardCoverage mP, mL, mN, covered, uncovered
        Select Case AwardOrderProblem(mP, mL, mN, who)
            Case "full-first": coverTxt = vbLf & "NOT ALLOWED: " & who & " signed up for the whole shift and comes first"
            Case "filled": coverTxt = vbLf & "NOT ALLOWED: every position already has a full-shift award"
            Case "over-hours": coverTxt = vbLf & "NOT ALLOWED: the partial awards add up to more than the hours left"
            Case Else
                If uncovered > 0 And covered > 0 Then
                    coverTxt = vbLf & "Covered " & HoursText(covered) & " h; " & HoursText(uncovered) & " h not covered (nobody charged for them)"
                End If
        End Select
        C("lblSummary").Caption = "Awarded " & nAw & " of " & mP.Positions & "   |   last awarded in call order: #" & _
            maxRank & " (" & Format$(maxHrs, "#,##0") & " hrs)" & vbLf & _
            "Charged (didn't sign up, ahead of #" & maxRank & "): " & nCh & " operator(s), " & HoursText(tot) & " hrs total" & coverTxt
    End If
End Sub

Private Sub lstEmployees_DblClick(ByVal Cancel As MSForms.ReturnBoolean)
    ToggleSigned
End Sub

Private Sub btnSignedUp_Click()
    ToggleSigned
End Sub

Private Sub btnPassed_Click()           ' AWARD toggle
    Dim i As Long: i = lstEmployees.ListIndex + 1
    If i < 1 Or i > mN Then MsgBox "Select a name first.", vbInformation: Exit Sub
    SetAwarded mL, i, Not mL(i).Awarded              ' v25: the rule is modPostings.SetAwarded
    mDirty = True
    RefreshList i - 1
End Sub

Private Sub ToggleSigned()
    Dim i As Long: i = lstEmployees.ListIndex + 1
    If i < 1 Or i > mN Then Exit Sub
    SetSignedUp mL, i, Not mL(i).SignedUp            ' v25: the rule is modPostings.SetSignedUp
    mDirty = True
    RefreshList i - 1
End Sub

' PARTIAL HOURS (v24): the selected person signed up for only part of the shift.
' Blank or the posting's full hours = the whole shift.
Private Sub mPartial_Click()
    Dim i As Long: i = lstEmployees.ListIndex + 1
    If i < 1 Or i > mN Then MsgBox "Select a name first.", vbInformation: Exit Sub
    Dim a As Variant, h As Double
    a = Application.InputBox(Prompt:="Hours " & mL(i).EmpName & " signed up for." & vbNewLine & vbNewLine & _
                             "The posting is " & mP.Hours & " h. Enter fewer hours for a partial sign-up, " & _
                             "or leave it blank for the whole shift.", _
                             Title:="Partial hours", Default:=IIf(mL(i).SignupHrs > 0, CStr(mL(i).SignupHrs), ""), Type:=2)
    If VarType(a) = vbBoolean Then Exit Sub                 ' Cancel
    Dim why As String                                       ' v25: the rule is modPostings.SetPartialHours
    If Not SetPartialHours(mL, i, CStr(a), mP.Hours, why) Then MsgBox why, vbExclamation, "Partial hours": Exit Sub
    h = mL(i).SignupHrs
    mDirty = True
    RefreshList i - 1
End Sub


' ================================================================
' SUBMIT
' ================================================================
Private Sub btnSubmit_Click()
    If mN = 0 Then Exit Sub
    If SubmitAward(mP, mL, mN) Then
        Finished mP.ID & " awarded."
    Else
        RefreshList lstEmployees.ListIndex
    End If
End Sub

Private Sub mNoTakers_Click()
    If mN = 0 Then Exit Sub
    If SubmitNoTakers(mP, mL, mN) Then
        Finished mP.ID & " closed - no takers."
    Else
        RefreshList lstEmployees.ListIndex
    End If
End Sub

Private Sub mOutside_Click()
    If mN = 0 Then Exit Sub
    Dim ot As Double, why As String                         ' v25: the same rule as a partial sign-up (PartialHoursValue)
    why = PartialHoursValue(C("txtOutHrs").Text, mP.Hours, ot)
    If why <> "" Then MsgBox why, vbExclamation, "Out of department": Exit Sub
    If SubmitAwardOutside(mP, mL, mN, C("cmbOutside").Text, ot) Then
        Finished mP.ID & " awarded out of department."
    Else
        RefreshList lstEmployees.ListIndex
    End If
End Sub

Private Sub mCancelReq_Click()
    If mN = 0 Then Exit Sub
    If SubmitCancel(mP, mL, mN) Then
        Finished mP.ID & " cancelled - no charges."
    Else
        RefreshList lstEmployees.ListIndex
    End If
End Sub

Private Sub Finished(ByVal msg As String)
    mDirty = False
    mN = 0
    lstEmployees.Clear
    C("lblInfo").Caption = "Pick a posting and click LOAD."
    C("lblSummary").Caption = ""
    SetActionButtons False
    LoadChoices
    C("lblStatus").Caption = msg & SaveWarnText() & "   " & C("lblStatus").Caption    ' v26.5.5 (A1): a failed save is named on the line
End Sub

Private Sub SetActionButtons(ByVal on_ As Boolean)
    ' v25.5 (bundle item 6): a supervisor awards to the list only; No Takers, Award Out of Dept and Cancel
    ' Request are greyed for a non-admin login (the three Submit* carry RequireAdmin as well)
    Dim adm As Boolean: adm = on_ And IsAdmin()
    btnSignedUp.Enabled = on_
    btnPassed.Enabled = on_
    mPartial.Enabled = on_
    btnSubmit.Enabled = on_
    mNoTakers.Enabled = adm
    mCancelReq.Enabled = adm
    mOutside.Enabled = adm
    C("cmbOutside").Enabled = adm
    C("txtOutHrs").Enabled = adm
    If Not on_ Then C("cmbOutside").Text = "": C("txtOutHrs").Text = ""
End Sub


' ================================================================
' WARN ON CLOSE
' ================================================================
Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If gErrorClosing Then Exit Sub                 ' v24.2: closed by the error handler, which says what happened
    If mDirty Then
        If MsgBox("Close without submitting?", vbYesNo + vbExclamation, "Unsaved") = vbNo Then Cancel = 1
    End If
End Sub
